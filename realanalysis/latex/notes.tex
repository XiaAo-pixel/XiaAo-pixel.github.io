\chapline{1. 导言：Riemann 积分的局限性}

\textbf{prop.} $f \in R[a,b] \iff f$ 在 $[a,b]$ 上几乎处处连续.

\textbf{Problems:}

（1）某些简单函数并非 Riemann 可积.

eg.（Dirichlet）$D(x) \notin R[0,1]$.

（2）Newton-Leibniz 公式中有额外限制.

$f \in C'[a,b]$，$\int_a^b f' = f(b) - f(a)$.

（3）积分与极限的可交换性.

eg. $\exists f_n \in C[0,1]$，$f_n$ 对 $n$ 单调收敛到 $f$.

而 $f \notin R[0,1]$.

（4）多元下存在可积性问题.

eg.（Besicovitch function）

$\exists$ 二元 R 可积 $f(x,y)$，对 $\forall$ 坐标系 $\{0\vec{x}, 0\vec{y}\}$.

$\exists x_0$，s.t. $f(x_0, y)$ 对 $y$ R 不可积.

（5）$R[a,b]$ 函数空间不完备.

$\mathrm{dist}(f,g) := \int_a^b \vert f-g\vert$. for $f, g$ in $R[a,b]$.

$\exists$ Cauchy 列 $\{f_n\}$，$\forall f \in R[a,b]$，$\mathrm{dist}(f_n, f) \to 0$.

\secn{2. 基本集合论}

\textbf{def.}（limit of sets）

 \textcircled{1}  设 $A_1 \supset A_2 \supset \cdots \supset A_n \supset \cdots$，定义 $\lim_{n\to\infty} A_n = \bigcap_{n=1}^{+\infty} A_n$.

Rem. $\lim 1_{A_n}(x) = 1_{\bigcap A_n}(x)$  pointwise

 \textcircled{2}  设 $X_1, X_2, \cdots, X_n, \cdots$ 为集族.

$\overline{\lim} X_n := \bigcap_{n=1}^{\infty} \bigcup_{k \ge n} X_k, \qquad \underline{\lim} X_n := \bigcup_{n=1}^{\infty} \bigcap_{k \ge n} X_k.$

us. $\overline{\lim} X_n \supset \underline{\lim} X_n$.

若 $\overline{\lim} X_n = \underline{\lim} X_n$，则称 $\lim X_n := \overline{\lim} X_n = \underline{\lim} X_n$.

Rem. $\overline{\lim} 1_{X_n}(x) := 1_{\overline{\lim} X_n}(x)$，$\underline{\lim} 1_{X_n}(x) := 1_{\underline{\lim} X_n}(x)$  pointwise

\textbf{prop.}（mapping over operation）

（1）$\bigcup_{\alpha \in A} f^{-1}(X_\alpha) = f^{-1}\Big(\bigcup_{\alpha \in A} X_\alpha\Big)$，$\bigcap_{\alpha \in A} f^{-1}(X_\alpha) = f^{-1}\Big(\bigcap_{\alpha \in A} X_\alpha\Big)$

$f^{-1}(B)^c = f^{-1}(B^c)$

（2）$\bigcup_{\alpha \in A} f(X_\alpha) = f\Big(\bigcup_{\alpha \in A} X_\alpha\Big)$，$\bigcap_{\alpha \in A} f(X_\alpha) \supset f\Big(\bigcap_{\alpha \in A} X_\alpha\Big)$

eg. $f: \{0,1\} \to \{1\}$. $f(\{0\}) \cap f(\{1\}) = \{1\}$，$f(\{0\}\cap\{1\}) = \emptyset$.

\textbf{def.}（Power set）$\mathcal{P}(X) := \{A : A \subseteq X\}$（also $2^X$）

\textbf{Th.}（fixed point of monotonic mapping）

$f: \mathcal{P}(X) \to \mathcal{P}(X)$，s.t. $A \subset B \Rightarrow f(A) \subset f(B)$.

assert. $\exists T \subset \mathcal{P}(X)$，s.t. $f(T) = T$.

pf. define $S := \{A \subset \mathcal{P}(X) : A \subset f(A)\}$. $T := \bigcup_{A \in S} A$.

$f(T) = \bigcup_{A \in S} f(A) \supset \bigcup_{A \in S} A = T \quad \Rightarrow \quad T \subset f(T)$

$\Rightarrow T \in S$，$\Rightarrow T \subset f(T)$

$\Rightarrow f(T) \subset f(f(T)) \Rightarrow f(T) \in S.$

$\Rightarrow f(T) \in \bigcup_{A \in S} A = T$

$\Rightarrow T = f(T). \qquad \square$

\textbf{Lemma.}（单调映射不动点原理）

设 $\mathscr{A}$ 是 $X$ 上的一个集族 s.t.

（1）$\mathscr{A}$ 对任意交封闭

（2）$\bigcup_{A \in \mathscr{A}} A \in \mathscr{A}$.

设 $f: \mathscr{A} \to \mathscr{A}$ s.t. $A \subset B \Rightarrow f(A) \subset f(B)$，则

$\exists T \in \mathscr{A},\ \text{s.t.}\ f(T) = T.$

pf. 定义 $S := \{A \in \mathscr{A} : f(A) \subset A\}$.（$S$ 非空，因 $f\big(\bigcup_{A \in \mathscr{A}} A\big) \subset \bigcup_{A \in \mathscr{A}} A$）

令 $T = \bigcap_{A \in S} A$，则

$f(T) = f\Big(\bigcap_{A \in S} A\Big) \subset \bigcap_{A \in S} f(A) \subset \bigcap_{A \in S} A = T$

故 $T \in S$.

则 $f(f(T)) \subset f(T)$，则 $f(T) \in S$.

从而 $T \subset f(T)$，于是 $T = f(T)$. $\square$

\secn{3. Cardinality of set}

\textbf{def.}（等势）

设 $A, B$ 为集合. 若 $\exists$ 双射 $f: A \to B$，则称 $A$ 与 $B$ 等势，记 $A \sim B$.

\textbf{e.g.} 设 $A$ 为集合.

（1）若 $A \sim \{1, 2, \cdots, n\}$，则记 $\mathrm{Card}(A) = n$.

（2）若 $A \sim \mathbb{N}$，则记 $\mathrm{Card}(A) = \aleph_0$（阿列夫零）.

（3）若 $A \sim \mathbb{R}$，则记 $\mathrm{Card}(A) = c$（连续统）.

\textbf{Rem.} $\overline{X} \le \overline{Y} \ \stackrel{\text{def}}{\iff}\ \exists$ 单射 $f: X \to Y$

$\overline{Y} \ge \overline{X} \ \stackrel{\text{def}}{\iff}\ \exists$ 满射 $g: Y \to X$

$\overline{X} = \overline{Y} \ \stackrel{\text{def}}{\iff}\ \exists$ 双射 $h: X \to Y$

\textbf{Th.}（Cantor–Bernstein）

若 $\exists$ 单射 $f: A \to B$，$g: B \to A$，则 $\overline{A} = \overline{B}$.

Tarski's proof.（用单调映射不动点原理）

设 $f: X \to Y$ 单，$g: Y \to X$ 单，

定义 $h: \mathcal{P}(X) \to \mathcal{P}(X)$，$h(A) = X \setminus g(Y \setminus f(A))$.

则 $A \subset B \Rightarrow h(A) \subset h(B)$.

故 $\exists T \in \mathcal{P}(X)$，s.t. $T = h(T)$.

则 $X \setminus T = g(Y \setminus f(T))$.

定义 $\widetilde{f}: X \to Y$，$\widetilde{f}(x) = \begin{cases} f(x), & x \in T \\ g^{-1}(x), & x \in X \setminus T \end{cases}$

易验证 $\widetilde{f}$ 是双射. 从而 $\overline{X} = \overline{Y}$. $\square$

\textbf{Def.}（偏序集 poset）

一个二元组 $(X, \leq)$ 被称为 poset，若

（1）$x \leq x$，$\forall x \in X$.

（2）$x \leq y$，$y \leq x \Rightarrow x = y$.

（3）$x \leq y$，$y \leq z \Rightarrow x \leq z$.

若 $\forall x, y \in X$，总有 $x \leq y$ 或 $y \leq x$，则称 $(X, \leq)$ 为全序集.

称 $x$ 是 $A \subset X$ 的上界，若 $a \leq x$，$\forall a \in A$.

称 $x$ 是 $X$ 的极大元，若 $\forall x' \in X$，$x \leq x' \Rightarrow x = x'$.

\textbf{Ax.}（选择公理）

任给一族非空集 $\{X_\alpha\}_{\alpha \in A}$（$A \neq \emptyset$，$X_\alpha \neq \emptyset$），则

$\exists f: A \to \bigcup_{\alpha \in A} X_\alpha,\ \text{s.t.}\ f(\alpha) \in X_\alpha.$

等价形式（Zorn's lemma）

若 poset $(X, \leq)$ 的每个全序子集有上界，则 $X$ 有极大元.

\textbf{prop.} 当 $X, Y \neq \emptyset$ 时，$\overline{X} \leq \overline{Y} \iff \overline{Y} \geq \overline{X}$.

pf. "$\Rightarrow$" 设 $f: X \to Y$ 单射，定义 $g: Y \to X$，

\[
g(y) = \begin{cases} f^{-1}(y), & y \in f(X) \\ x_0, & y \in Y \setminus f(X) \end{cases}
\]

则 $g$ 为满射.

"$\Leftarrow$" 设 $g: Y \to X$ 满射，定义

$f(x) \in g^{-1}(x),\ \forall x \in X$.（选择公理）

则 $f$ 为单射. $\square$

\textbf{prop.} 对任意非空 $X, Y$，总有 $\overline{X} \leq \overline{Y}$ 或 $\overline{Y} \leq \overline{X}$.

pf. 定义 $\mathscr{X} = \{(A, f) : A \subset X,\ f: A \to Y\ \text{单}\}$. $\mathscr{X} \neq \emptyset$.

称 $(A, f) \leq (B, h)$，若 $A \subset B$ 且 $h\vert_A = f$.

则 $(\mathscr{X}, \leq)$ 是 poset.

对 $\forall$ 全序子集 $\{(A_\alpha, f_\alpha)\} \subset \mathscr{X}$，则 $\big(\bigcup A_\alpha,\ \overline{f}\big)$ 是其上界.

其中 $\overline{f}(x) = f_\alpha(x)$ 若 $x \in A_\alpha$.

由 Zorn's lemma，$\mathscr{X}$ 有极大元，记 $(A, f)$.

下证 $A = X$ 或 $f(A) = Y$.

否则，$A \subsetneqq X$ 且 $f(A) \subsetneqq Y$.

取 $x_0 \in X \setminus A$，$y_0 \in Y \setminus f(A)$.

令 $\widehat{A} = A \cup \{x_0\}$，$\widehat{f}(x) = \begin{cases} f(x), & x \in A \\ y_0, & x = x_0 \end{cases}$

则 $(A, f) \leq (\widehat{A}, \widehat{f})$ 且 $(A, f) \neq (\widehat{A}, \widehat{f})$，与 $(A, f)$ 极大矛盾.

故 $A = X$ 或 $f(A) = Y$.

而 $f(A) = Y \Rightarrow \exists\, g(y) = f^{-1}(y)$ 单.

故 $\overline{X} \leq \overline{Y}$ 或 $\overline{Y} \leq \overline{X}$. $\square$

\textbf{prop.}（无最大基数）$\overline{X} < \overline{\overline{\mathcal{P}(X)}}$

pf. 反设 $\exists$ 双射 $f: X \to \mathcal{P}(X)$.

记 $A = \{x \in X : x \notin f(x)\} \subset \mathcal{P}(X)$.

设 $\exists a \in X$，s.t. $A = f(a)$.

若 $a \in A$，则 $a \in f(a) \Rightarrow a \notin A$.

若 $a \notin A$，则 $a \notin f(a) \Rightarrow a \in A$. $\square$

\textbf{def.} 若 $\overline{X} \leq \overline{\overline{\mathbb{N}}}$，则称 $X$ 是可数集.

\textbf{prop.}（1）设 $X, Y$ 可数，则 $X \times Y$ 可数.

（2）设 $A, X_\alpha$ 可数，则 $\bigcup_{\alpha \in A} X_\alpha$ 可数.

\textbf{e.g.} 若 $f: \mathbb{R} \to \mathbb{R}$ 单增，则 $f$ 的间断点集可数.

pf. 设 $S = \{x : x$ 是 $f$ 的不连续点$\}$.

$\forall x \in S$，$(f(x-0), f(x+0))$ 是开区间，且两两不交.

故 $\{(f(x-0), f(x+0)) : x \in S\}$ 可数.

故 $S$ 可数.

\textbf{e.g.} 若 $f: \mathbb{R} \to \mathbb{R}$ 下凸，则 $f$ 的不可微点可数.

pf. $f$ 下凸 $\Rightarrow f'_-(x) \uparrow$，$f'_+(x) \uparrow$

令 $S = \{x \in \mathbb{R} : f'_-(x) < f'_+(x)\}$ 是 $f$ 的不可微点集.

$\forall x \in S$，$(f'_-(x), f'_+(x))$ 两两不交，故 $S$ 可数.

\textbf{e.g.} $\overline{\mathbb{R}} = \overline{\overline{\mathcal{P}(\mathbb{N})}}$

pf. $\mathcal{P}(\mathbb{N}) = \{\mathbb{N}$ 的有限子集$\} \sqcup \{\mathbb{N}$ 的无限子集$\}$.

若 $A \subset \mathbb{N}$ 无限，定义 $\chi_A = \sum_{i \in A} \frac{1}{2^i}$.

\secn{4. 常用集合簇}

\textbf{def.}（Topology）

称 $\mathcal{T}$ 是 $X$ 上的拓扑，若 $\mathcal{T} \subset \mathcal{P}(X)$ 满足：

（1）$X, \emptyset \in \mathcal{T}$

（2）$A \cap B \in \mathcal{T}$，若 $A, B \in \mathcal{T}$.

（3）$\mathcal{T}$ 对任意并封闭.

\textbf{e.g.}（Euclidean Topology）

开球 $B_\delta(x) := \{y \in \mathbb{R}^n : \vert y - x\vert < \delta\}$.

$\{B_\delta(x) : x \in \mathbb{R}^n,\ \delta > 0\}$ 生成的拓扑称为欧氏拓扑 $\mathcal{T}$.

称 $\mathcal{T}$ 中的元素为开集.

e.g. $\mathbb{R}$ 上的任一开集是由开区间的可数并产生.

\textbf{def.} $F_\sigma$ 集 = 闭集的可数并.

$G_\delta$ 集 = 开集的可数交.

\textbf{def.}（$\sigma$ 代数）

称 $\mathscr{A}$ 是非空集 $X$ 上的 $\sigma$ 代数，若

（1）$\emptyset \in \mathscr{A}$

（2）$A \in \mathscr{A} \Rightarrow A^c \in \mathscr{A}$

（3）对可数并封闭.

\textbf{prop.}（1）$X \in \mathscr{A}$

（2）$A, B \in \mathscr{A} \Rightarrow A \setminus B \in \mathscr{A}$

（3）对可数交封闭.

\textbf{def.}（Borel $\sigma$ 代数）

包含开集簇 $\mathcal{T}$ 的最小 $\sigma$ 代数，即 $\bigcap_{\mathscr{A} \supset \mathcal{T}} \mathscr{A}$.

\textbf{e.g.} $G_\delta := \bigcap_{n=1}^{\infty} O_n$，$O_n$ 是开集.

$f: \mathbb{R}^n \to \mathbb{R}$，$U = \{f$ 的连续点$\}$ 是 $G_\delta$.

\[
U = \bigcap_{k=1}^{\infty} \Big\{ x \in \mathbb{R}^n \ \Big|\ \lim_{\delta \to 0} \sup_{x_1, x_2 \in B_\delta(x)} \vert f(x_1) - f(x_2)\vert < \tfrac{1}{k} \Big\}
\]

\textbf{e.g.} $F_\sigma := \bigcup_{n=1}^{\infty} E_n$，$E_n$ 是闭集.

$f: \mathbb{R}^n \to \mathbb{R}$，$U = \{f$ 的不连续点$\}$ 是 $F_\sigma$.（$F_\sigma = G_\delta^c$）

\secn{5. Cantor 集，Cantor 函数}

设 $F_0 = [0, 1]$.

step 1：去掉 $F_0$ 的中央开区间 $(\tfrac{1}{3}, \tfrac{2}{3})$

得到 $F_1 = [0, \tfrac{1}{3}] \cup [\tfrac{2}{3}, 1] := F_{1,1} \cup F_{1,2}$.

step 2：去掉 $F_{1,1}, F_{1,2}$ 的中央开区间

得到 $F_2 := F_{2,1} \cup F_{2,2} \cup F_{2,3} \cup F_{2,4}$.

……

step $k$：得到 $F_k := F_{k,1} \cup F_{k,2} \cup \cdots \cup F_{k,2^k}$.

于是得到一列 $\{F_k\}$ 递减，定义

Cantor 集 $C = \lim_{k \to \infty} F_k = \bigcap_{k=1}^{\infty} F_k$.

\textbf{Rem.} $C$ 中任一元素 $x$ 能一一表示成 $\sum_{i=1}^{\infty} \frac{b_i}{3^i}$，$b_i \in \{0, 2\}$.

\textbf{prop.}（1）$C$ 为非空有界闭集.

（2）$C$ 无内点.（$mC = 0$）

（3）$C$ 为 perfect set（$C' = C$）

（4）$\overline{\overline{C}} = \overline{\overline{\mathbb{R}}} = c$.

pf.（1）$F_k \supset F_{k+1}$，且 $F_k$ 均非空有界闭，由闭集套可知 $C$ 非空.

又闭集对任意交封闭，故 $C$ 是闭集.

（2）反设 $x \in C = \bigcap_{k=1}^{\infty} F_k$ 为内点，则

$\exists \delta > 0,\ \text{s.t.}\ U_\delta(x) \subset C$

又 $\forall k$，$x \in F_k$，故 $\vert U_\delta(x)\vert \leq \vert F_k\vert \Rightarrow 2\delta \leq (\tfrac{2}{3})^k,\ \forall k$，矛盾.

（3）因 $C$ 是闭集，故 $C' \subset C$. 只需 $C \subset C'$.

$\forall x_0 \in C$，$x_0$ 可表成 $\sum_{i=1}^{\infty} \frac{b_i}{3^i}$，$b_i \in \{0, 2\}$.

令 $x_n = \sum_{i=1}^{n} \frac{b_i}{3^i} + \sum_{n < i \leq 2n} \frac{2}{3^i}$，则 $\{x_n\}$ 为 $C$ 中互异点列.

又 $x_n \to x_0$，故 $x_0 \in C'$. 则 $C \subset C'$.

（4）给出满射

\[
\varphi:\quad \begin{aligned} C &\to [0,1] \\ \sum_{i} 2\frac{a_i}{3^i} &\mapsto \sum_{i=1}^{\infty} \frac{a_i}{2^i} \end{aligned} \qquad a_i \in \{0,1\}.
\]

从而 $\overline{\overline{C}} \geq \overline{\overline{[0,1]}}$. 又 $C \subset [0,1]$，易知 $\overline{\overline{C}} \leq \overline{\overline{[0,1]}}$.

于是 $\overline{\overline{C}} = \overline{\overline{[0,1]}} = \overline{\overline{\mathbb{R}}}$. $\square$

\textbf{def.}（Cantor function 两种定义）

（1）先定义阶梯函数 $f: [0,1] \setminus C \to [0,1]$. 单调不减.

$\widetilde{f}: [0,1] \to [0,1]$，s.t. $\widetilde{f}\big\vert_{[0,1] \setminus C} = f$.

$\forall x \in C$，$\widetilde{f}(x) := \lim_{y \to x} f(y)$.

\[
\widetilde{f}(x) = \sup_{\substack{y < x \\ y \in [0,1] \setminus C}} f(y) = \inf_{\substack{y > x \\ y \in [0,1] \setminus C}} f(y)
\]

（2）先定义 $g: C \to [0,1]$

\[
x = \sum_{i=1}^{\infty} \frac{k_i}{3^i} \ \longmapsto\ \sum_{i=1}^{\infty} \frac{k_i}{2^{i+1}}.
\]

延拓 $\widetilde{g}(x) = \sup_{\substack{y < x \\ y \in C}} g(y)$.

\textbf{Rem.} $\widetilde{f} = \widetilde{g}$，均称为 Cantor function.

\secn{6. Baire 纲集}

\textbf{Th.}（Baire Category）

设 $E \subset \mathbb{R}^n$ 非空，$E$ 是 $F_\sigma$ 集 即 $E = \bigcup_{n=1}^{\infty} F_n$，其中 $F_n$ 闭.

若 $F_n$ 无内点（$F_n^\circ = \emptyset$），则 $E$ 也无内点（$E^\circ = \emptyset$）.

pf. $\forall$ 开球 $B_{\delta_1}(x_1) \not\subset F_1$，

取 $x_2 \in B_{\delta_1}(x_1) \setminus F_1$，

则 $\exists \delta_2 < \tfrac{1}{2} \delta_1$，s.t. $\overline{B_{\delta_2}(x_2)} \subset B_{\delta_1}(x_1) \setminus F_1$.

取 $x_3 \in B_{\delta_2}(x_2) \setminus F_2 \neq \emptyset$，

类似地，$\exists \delta_3 < \tfrac{1}{2} \delta_2$，s.t. $\overline{B_{\delta_3}(x_3)} \subset B_{\delta_2}(x_2) \setminus F_2$.

递推地，$\exists\ \overline{B_{\delta_n}(x_n)} \subset B_{\delta_{n-1}}(x_{n-1}) \setminus F_{n-1}$，$\delta_n \to 0$.

$\exists x_*$，s.t. $\{x_*\} = \bigcap_{n=1}^{\infty} \overline{B_{\delta_n}(x_n)} = \bigcap_{n=1}^{\infty} B_{\delta_n}(x_n)$.

但 $x_* \in \overline{B_{\delta_n}(x_n)} \Rightarrow x_* \notin F_{n-1},\ \forall n$.

故 $x_* \notin \bigcup_{n=1}^{\infty} F_n = E$，此即 $B_{\delta_1}(x_1) \not\subset E$ 对任一开球 $B_{\delta_1}(x_1)$ 成立.

故 $E^\circ = \emptyset$. $\square$

\textbf{e.g.} $\mathbb{R}$ 不可数.

pf. 反设 $\mathbb{R}$ 可数，$\mathbb{R} = \bigcup_{k=1}^{\infty} \{x_k\}$ 是 $F_\sigma$ 集，且 $\{x_k\}$ 无内点.

由 Baire Thm，$\mathbb{R}$ 无内点. 矛盾. $\square$

\textbf{e.g.} 设 $f: \mathbb{R} \to \mathbb{R}$ 无穷次可微，且 $\forall x \in \mathbb{R}$，$\exists n$，s.t. $f^{(n)}(x) = 0$.

则 $\exists (\alpha, \beta) \subset \mathbb{R}$，s.t. $f\vert_{(\alpha,\beta)}$ 是多项式.

pf. $\mathbb{R} = \bigcup_{n=1}^{\infty} \{x : f^{(n)}(x) = 0\}$ 是 $F_\sigma$ 集.

由 Baire Thm，$\exists k$，s.t. $\{x : f^{(k)}(x) = 0\}$ 有内点.

即 $\exists (\alpha, \beta) \subset \{x : f^{(k)}(x) = 0\}$. $\square$

\textbf{def.}（第一纲集）

称 $E \subset \mathbb{R}^n$ nowhere dense（无处稠密，稀疏），若 $E$ 无内点.

称 $F$ 是第一纲集，若 $F = \bigcup_{n=1}^{\infty} E_n$，其中 $E_n$ 稀疏.

非第一纲集称为第二纲集.

\textbf{Th.}（Baire）第一纲集无内点.

\secn{7. 连续延拓}

\textbf{def.} 点到集合的距离 $d(x, E) := \inf_{y \in E} \vert x - y\vert$.

\textbf{Th.} 对确定的 $E$，$f: x \to d(x, E)$ 是 1-Lipschitz 的.

\textbf{lemma}（Urysohn 连续分离）

设 $A, B \subset \mathbb{R}^n$ 是不交非空闭集，则 $\forall a, b$，$\exists$ 连续 $f: \mathbb{R}^n \to [a,b]$，

s.t. $f^{-1}(a) = A$，$f^{-1}(b) = B$.

pf. 不妨设 $[a,b] = [0,1]$. 直接构造 $f(x) = \dfrac{d(x, A)}{d(x, A) + d(x, B)}$. $\square$

\textbf{Th.}（Tietze's extension）

设 $A \subset \mathbb{R}^n$ 非空闭，$f: A \to [-M, M]$ 连续.

则 $\exists$（不唯一）$\widetilde{f}: \mathbb{R}^n \to [-M, M]$ 连续，且 $\widetilde{f}\vert_A = f$.

pf. 作截断

\[
f_1(x) = \begin{cases} \dfrac{M}{3}, & f(x) \geq \dfrac{M}{3} \\[2mm] f(x), & -\dfrac{M}{3} \leq f(x) \leq \dfrac{M}{3} \\[2mm] -\dfrac{M}{3}, & f(x) \leq -\dfrac{M}{3} \end{cases} \qquad f_1(x) \in \Big[-\dfrac{M}{3}, \dfrac{M}{3}\Big].
\]

取 $A_1 = f_1^{-1}\big(-\dfrac{M}{3}\big)$，$B_1 = f_1^{-1}\big(\dfrac{M}{3}\big)$，则 $A_1 \cap B_1 = \emptyset$. 由 Urysohn lemma，

$\exists g_1: \mathbb{R}^n \to \big[-\dfrac{M}{3}, \dfrac{M}{3}\big]$，s.t. $g_1(A_1) = -\dfrac{M}{3}$，$g_1(B_1) = \dfrac{M}{3}$.

从而 $\vert f(x) - g_1(x)\vert < \dfrac{2}{3}M$. 对 $f - g_1$ 重复上述过程

$\exists g_k: \mathbb{R}^n \to \Big[-\dfrac{1}{3}\big(\tfrac{1}{3}\big)^{k-1}M,\ \dfrac{1}{3}\big(\tfrac{1}{3}\big)^{k-1}M\Big]$ 连续，s.t.

\[
\vert f(x) - g_1(x) - \cdots - g_k(x)\vert \leq \big(\tfrac{1}{3}\big)^k M.
\]

令 $\widetilde{f}(x) = \sum_{k=1}^{\infty} g_k(x)$ 一致收敛，$\sum \vert g_k(x)\vert \leq \sum \dfrac{1}{3}\big(\tfrac{1}{3}\big)^{k-1}M \leq M$

则 $\widetilde{f}: \mathbb{R}^n \to [-M, M]$ 连续. $\square$

\textbf{e.g.} 设 $E \subset \mathbb{R}^n$ 非空，则

$\exists$ 连续 $f_k: \mathbb{R}^n \to \mathbb{R}$，s.t. $\lim_{k \to \infty} f_k(x) = 1_E(x)$

$\iff E$ 是 $F_\sigma$ 集也是 $G_\delta$ 集.

pf. "$\Leftarrow$". $E = \bigcup_{k=1}^{\infty} F_k = \bigcap_{k=1}^{\infty} G_k$，且 $F_1 \subset F_2 \subset \cdots$ 是单增闭集列，

$G_1 \supset G_2 \supset \cdots$ 是单减开集列.

于是 $F_k \subset G_k$，从而 $F_k$ 与 $G_k^c$ 是不交闭集.

由 Urysohn lemma，$\exists f_k: \mathbb{R}^n \to [0,1]$，s.t.

$f_k^{-1}(0) = G_k^c$，$f_k^{-1}(1) = F_k$.

易验证 $\lim_{k \to \infty} f_k(x) = 1_E(x)$.

"$\Rightarrow$". 对确定的 $\delta \in (0,1)$，

\[
E = \bigcup_{n \geq 1} \bigcap_{k \geq n} \{x \in \mathbb{R}^n : f_k(x) \geq \delta\} \quad \text{是 } F_\sigma \text{ 集}.
\]

\[
E = \bigcap_{n \geq 1} \bigcup_{k \geq n} \{x \in \mathbb{R}^n : f_k(x) > \delta\} \quad \text{是 } G_\delta \text{ 集}. \qquad \square
\]

\secn{1. Lebesgue 测度动机}

希望找到"体积"的推广

$(0)\quad m: \mathcal{P}(\mathbb{R}^d) \to [0, \infty]$，s.t.

（1）$m\Big(\prod_{i=1}^{d} (a_i, b_i)\Big) = (b_1 - a_1)(b_2 - a_2)\cdots(b_d - a_d)$

（2）$m(A) = m(\varphi(A))$，$A$ 是任一刚体运动

（3）任意两两不交的集合列，$m\Big(\bigcup_n A_n\Big) = \sum_n m(A_n)$

事实上 (0)(1)(2)(3) 是不自洽的. 以 $d = 1$ 为例，

定义 $x \sim y$ iff $x - y \in \mathbb{Q}$. 商集 $\mathbb{R}/\sim = \{[x] : x \in \mathbb{R}\}$.

由选择公理，对 $\forall [x]$，取 $\overline{x} \in [x] \cap [0, 1)$，构成集合 $A$.

定义 $A + r := \{\overline{x} + r \bmod \mathbb{Z} : \overline{x} \in A\}$.

对 $\forall r_1 \neq r_2 \in \mathbb{Q} \cap [0,1]$，$A + r_1 \cap A + r_2 = \emptyset$.

（否则 $\exists a_i \in A$，$a_1 + r_1 = a_2 + r_2 \Rightarrow a_1 - a_2 = r_2 - r_1 \in \mathbb{Q}$）

同时，$\forall x \in \mathbb{R}$，$\exists [\overline{y}]$，s.t. $x \in [\overline{y}]$，$\overline{y} \in A$

$\Rightarrow x - \overline{y} \in \mathbb{Q}$. 记 $r = x - \overline{y} \bmod \mathbb{Z}$，则 $x \in A + r$.

从而 $\bigcup_{r \in \mathbb{Q} \cap [0,1]} A + r = [0, 1]$

\[
m\bigcup_{r \in \mathbb{Q} \cap [0,1]} A + r \ =\ \sum m(A + r) \ =\ \sum m(A) \ =\ \begin{cases} 0, & \text{if } m(A) = 0 \\ \infty, & \text{if } m(A) > 0 \end{cases}
\]

这与 $m[0,1] = 1$ 矛盾.

\secn{2. 外测度与 Lebesgue 测度}

\textbf{def.}（outer measure）

\[
m^*(A) := \inf_{\substack{\text{开盒 } I_j \\ \text{s.t. } \bigcup I_j \supset A}} \sum \vert I_j\vert \qquad \text{则 } m^*: \mathcal{P}(\mathbb{R}^d) \to [0, \infty].
\]

\textbf{e.g.} 设 $A$ 可数，则 $m^*A = 0$.

pf. $A = \{x_1, x_2, \cdots, x_n\}$. $\forall \varepsilon > 0$，$\exists$ 方体 $I_j \supset x_j$，$\vert I_j\vert < \dfrac{\varepsilon}{2^j}$.

$m^*A \leq \sum_{j \geq 1} \vert I_j\vert \leq \sum_{j \geq 1} \frac{\varepsilon}{2^j} \leq \varepsilon$. 故 $m^*A = 0$.

\textbf{prop}

（1）$\forall \delta > 0$，记 $m_\delta^*(A) = \sum \vert I_j\vert$，$\bigcup I_j \supset A$ 且 $I_j$ 棱长小于 $\delta$，

则 $\lim_{\delta \to 0^+} m_\delta^*(A) = m^*A$.

（2）$A \subset B \Rightarrow m^*A \leq m^*B$.

（3）$\forall \{A_j\} \subset \mathcal{P}(\mathbb{R}^d)$，$m^*\big(\bigcup A_j\big) \leq \sum m^*A_j$.

pf. $\forall \varepsilon > 0$，$\exists$ 开盒 $\{I_{j,k}\}_{k=1}^{\infty}$ 覆盖 $A_j$，s.t.

$m^*A_j \leq \sum \vert I_{j,k}\vert \leq m^*A_j + \dfrac{\varepsilon}{2^j}$.

因 $\bigcup_j A_j \subset \bigcup_k \bigcup_j I_{j,k}$，从而

\[
m^* \bigcup_j A_j \ \leq\ m^* \bigcup_k \bigcup_j I_{j,k} \ =\ \sum_k \sum_j \vert I_{j,k}\vert
\]

\[
\leq\ \sum_j m^*A_j + \sum_j \frac{\varepsilon}{2^j} \ =\ \sum_j m^*A_j + \varepsilon.
\]

由 $\varepsilon$ 任意性即有 $m^*\bigcup A_j \leq \sum m^*A_j$. $\square$

（4）若 $d(A, B) > 0$，即 $\inf_{a \in A,\, b \in B} \vert a - b\vert > 0$，则 $m^*(A \cup B) = m^*A + m^*B$.

pf. 只需 $m^*(A \cup B) \geq m^*A + m^*B$.

对 $\forall \varepsilon > 0$，存在一致小的方体 $\{I_j\}, \{I_{j'}\}$ 覆盖 $A, B$，

且 $I_j, I_{j'}$ 不交，棱长小于 $d(A, B)/2$. 于是 $\{I_j\} \cup \{I_{j'}\}$ 覆盖 $A \cup B$

$\varepsilon + m^*(A \cup B) \geq \sum \vert I_j\vert + \sum \vert I_{j'}\vert \geq m^*(A) + m^*(B)$.

$\square$

（5）$m^*A = \inf\limits_{\text{开集 } O \supset A} m^*O$

pf. 开方体簇是开集的拓扑集.

$\forall$ 开集 $O$，$O = \bigcup_{j=1}^{\infty} I_j$

由单调性：$m^*A \leq m^*O$，从而 $m^*A \leq \inf m^*O$.

另一方面，$\forall \varepsilon > 0$，$\exists \{I_j\}$ 覆盖 $A$，且

$\varepsilon + m^*A \geq \sum \vert I_j\vert \geq m^*O \geq \inf m^*O$

从而 $m^*A \geq \inf m^*O$.

故 $m^*A = \inf m^*O$. $\square$

\textbf{def}（Caratheodory 条件）$L$-可测定义1.

若 $A$ 满足：$\forall$ 测试集 $T \subset \mathbb{R}^d$，

\[
m^*T = m^*(T \cap A) + m^*(T \cap A^c)
\]

则称 $A$ 为 Lebesgue 可测集.

\textbf{Thm.} Lebesgue 可测集构成 $\sigma$ 代数.

pf. 记 $\mathscr{A} := \{A : A$ 是 L 可测集$\}$.

（1）$\emptyset \in \mathscr{A}$

（2）$A \in \mathscr{A} \Rightarrow A^c \in \mathscr{A}$.（卡氏条件）

（3）$\forall A, B \in \mathscr{A}$，$\forall T$，有

\[
\begin{aligned}
m^*(T) &= m^*(T \cap A) + m^*(T \cap A^c) \\
&= m^*(T \cap A \cap B) + m^*(T \cap A \cap B^c) \\
&\quad + m^*(T \cap A^c \cap B) + m^*(T \cap A^c \cap B^c) \\
&\geq m^*(T \cap (A \cup B)) + m^*(T \cap (A \cup B)^c)
\end{aligned}
\]

另一方面，$m^*(T) \leq m^*(T \cap (A \cup B)) + m^*(T \cap (A \cup B)^c)$

从而 $\forall T$，$m^*T = m^*(T \cap (A \cup B)) + m^*(T \cap (A \cup B)^c)$.

即 $A, B \in \mathscr{A} \Rightarrow A \cup B \in \mathscr{A}$.

下只需证 $\forall$ 两两不交 $\{A_j\}_{j=1}^{\infty} \subset \mathscr{A}$，$\bigcup A_j \in \mathscr{A}$.

记 $S_n := \bigsqcup_{j=1}^{n} A_j$，$S = \bigcup_{j=1}^{\infty} A_j$，$\forall T$ 有界.

\[
m^*T = m^*(T \cap S_n) + m^*(T \cap S_n^c)
\]

\[
m^*(T \cap S_n) = \sum_{j=1}^{n} m^*(T \cap A_j) \quad \text{（递归）}
\]

由 $m^*(T \cap S_n) \leq m^*(T \cap S) < \infty$，有 $\sum_{j=1}^{\infty} m^*(T \cap A_j)$ 收敛.

一方面，$m^*(T) \leq m^*(T \cap S) + m^*(T \cap S^c)$.

另一方面，$m^*(T) = m^*(T \cap S_n) + m^*(T \cap S_n^c)$

\[
\geq m^*(T \cap S_n) + m^*(T \cap S^c)
\]

\[
m^*\big((T \cap S) \setminus (T \cap S_n)\big) = m^*\Big(\bigsqcup_{j=n+1}^{\infty} S_j\Big) \quad \text{（} m^*(T \cap S_n) = m^*(T \cap S) \text{）}
\]

\[
\leq \sum_{j=n+1}^{\infty} m^*S_j \ \to\ 0.
\]

从而 $m^*(T) = m^*(T \cap S) + m^*(T \cap S^c)$.

（旁注：$A \subset B$，$m^*(B \setminus A) = 0$ 时，$m^*A \leq m^*B$，且 $m^*B = m^*(B \setminus A \cup A) \leq m^*A$.）

\textbf{Prop.} 外测度为 0 的集合 L 可测.

$\mathscr{A} \supset \{A \subset \mathbb{R}^d : m^*A = 0\}$.

pf. $\forall T$，$m^*T \leq m^*\, T \cap A + m^*\, T \cap A^c$.

$\leq m^*A + m^*T$.

$= m^*T$.

故等号成立. $\square$

\textbf{def} 定义 $m = m^*\vert_{\mathscr{A}} : \mathscr{A} \to [0, \infty]$，称为（Lebesgue）测度.

$m$ 在满足 $m^*$ 性质的基础上，满足可数可加性：

$\forall$ 两两不交的集合 $\{A_j\}_{j=1}^{\infty}$，$m(\bigsqcup A_j) = \sum m(A_j)$.

\secn{3. 测度的基本性质}

\textbf{Fact 1.} 设 $m^*$ 在不交集簇 $\{A_j\}_{j=1}^{\infty}$ 上有限可加，则可数可加.

pf.

\[
0 \leq \sum_{j=1}^{\infty} m^*(A_j) - m^*\Big(\bigsqcup_{j=1}^{\infty} A_j\Big) \leq \sum_{j=1}^{\infty} m^*(A_j) - m^*\Big(\bigsqcup_{j=1}^{n} A_j\Big)
\]

\[
= \sum_{j=1}^{\infty} m^*(A_j) - \sum_{j=1}^{n} m^*(A_j)
\]

不妨设 $m^*(A_j) < \infty$，则 RHS $\to 0$ ($n \to \infty$).

于是 $m^*\Big(\bigsqcup_{j=1}^{\infty} A_j\Big) = \sum_{j=1}^{\infty} m^*(A_j)$.$ \qquad \square$

\textbf{Fact 2.} 设 $E_k \subset E_{k+1}$，$E = \bigcup E_k$，$A_k := E_{k+1} \setminus E_k$.

且 $\{A_k\}_{k=1}^{\infty} = P_1 \sqcup \cdots \sqcup P_n$ 可分解为有限部分. 若 $m^*$ 在每个子集簇 $P_j$ 上有限可加，则 $m^*E = \lim m^*E_k$.

pf. 不妨设 $\lim m^*E_k < \infty$.

\[
0 \leq m^*E - m^*E_k \ \leq\ m^*\ E \backslash E_k \ \leq\ \sum_{j \geq k} m^*A_j
\]

\[
= \sum_{i=1}^{n} \sum_{\substack{j \geq k \\ A_j \in P_i}} m^*A_j
\]

\[
\sum_{\substack{j \geq k \\ A_j \in P_i}} m^*A_j < \infty \quad \Longleftarrow \quad \sum_{\substack{j \leq k \\ A_j \in P_i}} m^*A_j = m^*(\bigcup A_j) \leq m^*E_{k+1}
\]

RHS $\to 0$（$k \to \infty$）.（from Fact 1.）则 $m^*E = \lim m^*E_k$. $\square$

\textbf{Lemma.}（卡氏引理）

设 $G \subset \mathbb{R}^n$ 是开集，$E \subset G$，定义 $E_k = \{x \in E \mid d(x, G^c) \geq \tfrac{1}{k}\}$.

则 $\lim m^*E_k = m^*E$.

pf. $E_k \subset E_{k+1}$，$A_k := E_{k+1} \setminus E_k$，$E = \bigcup E_k$.

若 $\vert i - j\vert \geq 2$，则 $d(A_i, A_j) > 0$，

于是 $m^*(A_i \cup A_j) = m^*(A_i) + m^*(A_j) \qquad (\ast)$

令 $P_1 = \{A_{2j-1}\}$，$P_2 = \{A_{2j}\}$，则 $m^*$ 在 $P_1$、$P_2$ 上有限可加.

根据 Fact 2：$\lim m^*E_k = m^*E$. $\square$

\textbf{Prop.} $\forall$ 闭集 $F$，$F \in \mathscr{A}$.

pf. $\forall T \subset \mathbb{R}^n$，令 $E_k := \{x \in F^c \mid d(x, F) \geq \tfrac{1}{k}\}$.

$d(F, E_k) \geq \tfrac{1}{k}$，$d(F \cap T,\ E_k \cap T) \geq \tfrac{1}{k}$.

从而有有限可加性

\[
m^*\big(T \cap (F \cup E_k)\big) = m^*(T \cap F) + m^*(T \cap E_k).
\]

式中 $m^*(T \cap E_k) \to m^*(T \cap F^c)$（卡氏引理）

LHS $\leq m^*T$. 于是 $m^*T \geq m^*(T \cap F) + m^*(T \cap F^c)$.

从而 $m^*T = m^*(T \cap F) + m^*(T \cap F^c)$. $\square$

\textbf{Prop.} $\forall$ Borel set $G$，$G \in \mathscr{A}$. 即 $B \subset \mathscr{A}$.

pf. 闭集 $F \in \mathscr{A}$，且 $\mathscr{A}$ 是 $\sigma$ 代数 $\Rightarrow$ 开集 $G \in \mathscr{A}$.

$\Rightarrow$ Borel 集 $G \in \mathscr{A}$. $\square$

\secn{4. 测度的正则性}

\textbf{Prop.} $\forall E \in \mathscr{A}$，$\forall \varepsilon > 0$，

（1）$\exists$ 开集 $G \supset E$，s.t. $m(G \setminus E) < \varepsilon$.（外正则性）

（2）$\exists$ 闭集 $F \subset E$，s.t. $m(E \setminus F) < \varepsilon$.（内正则性）

pf.（1）若 $E \in \mathscr{A}$ 有界，且 $mE < \infty$，则 $\exists$ 开集 $G \supset E$，s.t.

$mE \leq mG \leq mE + \varepsilon$.

若 $E \in \mathscr{A}$ 无界，则 $\exists$ 开集 $G_k \supset E \cap B_k(0)$，s.t.

$m\big(G_k \setminus (E \cap B_k(0))\big) < \dfrac{\varepsilon}{2^k}$.

令 $G = \bigcup_{k=1}^{\infty} G_k$ 为开集，$G \supset E$ 且

$m(G \setminus E) \leq \sum m\big(G_k \setminus (E \cap B_k(0))\big) < \varepsilon$.

（2）对 $E^c$ 用（1），令 $F = G^c$ 即可. $\square$

\textbf{Prop}

（1）$\forall E \subset \mathbb{R}^n$，$\exists\ G_\delta$ 集 $G \supset E$，s.t. $m^*G = m^*E$.

（outer regularity for outer measure）

（2）$\forall E \in \mathscr{A}$，$\exists\ G_\delta$ 集 $G \supset E$，$\exists\ F_\sigma$ 集 $F \subset E$，s.t. $m(G \setminus F) = 0$.

（regularity of L-measure）

pf.（1）不妨 $m^*E < \infty$，$\exists$ 开集 $G_k \supset E$，s.t. $m^*E < m^*G_k + \tfrac{1}{k}$.

令 $G = \bigcap_{k=1}^{\infty} G_k$ 是 $G_\delta$ 集，$G \supset E$ 且 $m^*E \leq mG \leq mG_k \leq m^*E + \tfrac{1}{k}$

令 $k \to \infty$，则 $m^*E = m^*G$.

（2）$\exists G_k \supset E \supset F_k$，s.t. $m(G_k \setminus E) < \tfrac{1}{k}$，$m(E \setminus F_k) < \tfrac{1}{k}$.

$G := \bigcap G_k$，$F := \bigcup F_k$，则

$0 \leq m(G \setminus F) \leq m(G_k \setminus F_k) < \dfrac{2}{k} \to 0$. $\square$

\textbf{Thm.} $\forall A \in \mathscr{A}$，$\exists$ Borel 集 $F \in B$，零测集 $O \in \mathcal{O}$，s.t. $A = F \cup O$.

即 $\mathscr{A} = \sigma(B \cup \mathcal{O})$.

\textbf{Prop.} 若 $A \in \mathscr{A}$，则 $mA = \sup\limits_{\text{闭} F \subset A} mF = \inf\limits_{\text{开} G \supset A} mG$.

\textbf{Prop}

（1）设可测集列 $E_k \subset E_{k+1}$，则

$\lim mE_k = m \lim E_k$.

（2）设可测集列 $E_k \supset E_{k+1}$，且 $E_1 < \infty$，则

$\lim mE_k = m \lim E_k$.

（3）设 $\{E_k\}$ 是可测集列，则

\[
m\ \underline{\lim}\ E_k \ \leq\ \underline{\lim}\ mE_k \ \leq\ \overline{\lim}\ mE_k \ \leq\ m\ \overline{\lim}\ E_k.
\]

最后一个不等号需在条件 $m \bigcup E_k < \infty$ 下成立.（ \textcircled{1} ）

\textbf{Rem.} $\overline{\lim} E_k = \{x \in E_k$ for infinitely many $k\}$

$\underline{\lim} E_k = \{x \in E_k$ for all but finitely many $k\}$

proof of（ \textcircled{1} ）. 考虑 $E_k \subset \bigcup_{j \geq k} E_j$.

\[
\overline{\lim}\ m E_k \ \leq\ \lim m \bigcup_{j \geq k} E_j \ =\ m \lim \bigcup_{j \geq k} E_j \ =\ m \bigcap_{k=1}^{\infty} \bigcup_{j \geq k} E_j = m\ \overline{\lim}\ E_k.
\]

\textbf{Prop} 设 $K \subset \mathbb{R}^n$ 是紧集（有界闭），记 $K_\delta = \{x \in \mathbb{R}^n \mid d(x, K) < \delta\}$，

则 $mK = \lim\limits_{\delta \to 0} mK_\delta$.

pf. $K \subset K_{\delta_1} \subset K_{\delta_2}$. 于是 $mK \leq \lim\limits_{\delta \to 0} mK_\delta$.

只需证 $mK \geq \lim\limits_{j \to \infty} mE_j$，$E_j := K_{\frac{1}{j}}$.

$mE_1 < \infty$，$E_j \supset E_{j+1}$，从而 $\lim mE_j = m \lim E_j$.

下证 $K = \lim E_j$，即 $K = \lim K_\delta$.

设 $K \neq \lim K\delta$，则 $\exists x \in \bigcap K\delta \setminus K$，

取 $\overline{\delta} = d(x, K) > 0$，则 $x \notin K\delta$，当 $\delta < \overline{\delta}$ 时. 矛盾. $\square$

\textbf{Prop.}（内正则性）$\forall A \in \mathscr{A}$，$mA = \sup\limits_{\text{紧} K \subset A} mK$.

pf. $mA = \sup\limits_{\text{闭} F \subset A} mF$

$= \sup_F \sup_{r > 0} m\big(F \cap \overline{B_r(0)}\big)$

$\leq \sup mK \ \leq\ mA$.

故等号成立. 则 $mA = \sup mK$. $\square$

\secn{5. 正测度集的性质}

\textbf{Fact.} 设 $A_1, \cdots, A_k$ 可测，$A_i \subset B$. 若 $\sum mA_i > (k-1) m^*B$，则

$\bigcap A_k \neq \emptyset$.

pf. 反设 $\bigcap A_k = \emptyset$，则 $B$ 中每个点至多在 $k-1$ 个 $A_i$ 中.

\[
\sum_{i=1}^{k} mA_i = \sum_{I \subset \{1, \cdots, k\}} \vert I\vert \cdot mA_I \ \leq\ (k-1) \sum mA_I \ =\ (k-1)\ m \bigcup A_I \ \leq\ (k-1) m^*B.
\]

矛盾. 故 $\bigcap A_k \neq \emptyset$.

\textbf{Thm}（Steinhaus' Thm 1）

设 $A \subset \mathbb{R}^n$ 是正测集，$k \in \mathbb{Z}_+$，则 $\exists \delta > 0$，s.t. 只要 $\vert\vec{a}\vert < \delta$，则

\[
\bigcap_{j=1}^{k} (A + j\vec{a}) \neq \emptyset.
\]

（等价形式是 $\exists \vec{x} \in A$，s.t. $\{\vec{x},\ \vec{x}-\vec{a},\ \cdots,\ \vec{x}-k\vec{a}\} \subset A$）

pf. 取紧子集 $K \subset A$ 且 $mK > 0$，$K$ 存在是因 $0 < mA = \sup\limits_{\text{闭} F \subset A} mF$.（from Fact）

下证 $\bigcap_{j=1}^{p} (K + j\vec{a}) \neq \emptyset$ 对 $\vert\vec{a}\vert$ 足够小成立.

取 $\delta > 0$，s.t. $mK_\delta < \dfrac{p}{p-1} mK$，则只要 $\vert\vec{a}\vert < \dfrac{\delta}{p}$，则

\[
K + j\vec{a} \ \subset\ K_\delta
\]

且有 $\sum_{j=1}^{p} m(K + j\vec{a}) = p \cdot mK > (p-1)\ mK_\delta$.

由 Fact 知 $\bigcap_{j=1}^{p} (K + j\vec{a}) \neq \emptyset$. $\square$

\textbf{Thm}（Steinhaus' Thm 2）

设 $mA > 0$，则 0 的邻域 $\subset A - A := \{\overline{x} - \overline{y}\}$. $(\ast)$

pf. $(\ast)$ iff $A - A = \{a \in \mathbb{R}^n \mid A \cap (A + a) \neq \emptyset\}$.

由 Steinhaus' Thm 1，

$\exists \delta > 0$，只要 $\vert\vec{a}\vert < \delta$ 则上式成立. $\square$

\textbf{Lemma.} 若 $mA > 0$，则 $\forall \varepsilon \in (0, 1)$，$\exists$ 方体 $I$，s.t.

\[
\frac{m(I \cap A)}{mI} > 1 - \varepsilon.
\]

pf. by def，$mA = \inf\limits_{A \subset \bigcup I_j} \sum mI_j > 0$.

$\exists$ 方体列 $\{I_j\}$，$A \subset \bigcup I_j$，且

\[
1 - \varepsilon < \frac{mA}{\sum mI_j} \ \leq\ \frac{\sum m(A \cap I_j)}{\sum mI_j} \ \leq\ \sup_j \frac{m(A \cap I_j)}{mI_j}
\]

令 $I = I_j$ 即已得证. $\square$

\textbf{Rem.} 方体 $I$ 之顶点坐标可都是有理数，因 $\dfrac{m\, A \cap I}{mI}$ 关于 $I$ 的顶点坐标连续.

\textbf{Thm}（Steinhaus' Thm 3）

设 $mA,\ mB > 0$，则 $A - B$ 有内点.

pf. Step 1. $\exists$ box $Q$，$a' \in \mathbb{R}^n$，$Q' := Q + a'$，s.t.

\[
\frac{m\, B \cap Q'}{mQ'} > \frac{2}{3} \quad \text{且} \quad \frac{m\, A \cap Q}{mQ} > \frac{2}{3}
\]

由 Lemma 及后注可知，$\exists$ 有理方体 $I, J$，s.t.

\[
\frac{m\, A \cap I}{mI} > \frac{2}{3} \quad,\quad \frac{m\, B \cap J}{mJ} > \frac{2}{3}
\]

有理方体分划 $I = \bigcup\limits_{\text{有限}} Q_i$，$J = \bigcup\limits_{\text{有限}} Q_j'$，且 $Q_i$ 与 $Q_j'$ 平移重合.

\[
\frac{2}{3} < \frac{m\, A \cap I}{mI} = \frac{\sum m\, A \cap Q_i}{\sum m Q_i} \ \leq\ \max \frac{m\, A \cap Q_i}{mQ_i}
\]

同理 $\dfrac{2}{3} < \dfrac{m\, B \cap J}{mJ} \leq \max \dfrac{m\, B \cap Q_j'}{m Q_j'}$.

则取 $Q = Q_i$，$Q' = Q_j'$ 即可.

Step 2. $m\big((B + a') \cap A\big)$（构建 $A$ 与 $B$ 之间联系）

\[
\geq m\big((B \cap Q' + a') \cap (A \cap Q)\big)
\]

\[
= m(B \cap Q' + a') + m(A \cap Q) - m\big((B \cap Q' + a') \cup (A \cap Q)\big)
\]

\[
\geq m(B \cap Q') + m(A \cap Q) - mQ \geq \frac{1}{3} mQ > 0.
\]

Step 3. $A - B$ 有内点.

从 $D = (B + a') \cap A$，$mD > 0$ 出发，由 Steinhaus' Thm2，

$D - D$ 以 0 为内点，则

\[
A \cap (B + a') - A \cap (B + a')
\]

\[
\subset A - (B + a') = A - B - a'
\]

故 $A - B$ 以 $a'$ 为内点. $\square$

\textbf{Thm.}（Brunn–Minkowski Inequality）

设 $A, B$ 是正测集，则

\[
\big(m^*(A+B)\big)^{\frac{1}{d}} \ \geq\ (mA)^{\frac{1}{d}} + (mB)^{\frac{1}{d}}
\]

pf. Step 1. $A, B$ 是方体，则 $A + B$ 也是方体，亦可测.

此时 B-M 不等式即

\[
\big(\Pi (a_i + b_i)\big)^{\frac{1}{d}} \ \geq\ (\Pi a_i)^{\frac{1}{d}} + (\Pi b_i)^{\frac{1}{d}}
\]

Step 2. 设 $A, B$ 是有限个内部不交方体，对 $N_A + N_B$ 作归纳.

由 Step 1 知 $N_A + N_B \leq 2$ 时成立.

不妨设 $N_A \geq 2$，则存在超平面 $\{x \in \mathbb{R}^n \mid x_i = t\}$ 把 $A$ 中某两个方体隔开（不被切开）. 设该超平面为 $x_1 = 0$，

$A_+ := \{x \in \mathbb{R}^n \mid x_1 \geq 0\}$，$A_- := \{x \in \mathbb{R}^n \mid x_1 \leq 0\}$.

平移 $B$ 使得 $\dfrac{m(A_+)}{m(A)} = \dfrac{m(B_+)}{m(B)}$. $A_\pm + B_\pm \subset (A + B)_\pm$.

则 $\big(m(A+B)\big)^{\frac{1}{n}} = \big(m(A+B)_+ + m(A+B)_-\big)^{\frac{1}{n}}$

\[
\geq \big(m(A_+ + B_+) + m(A_- + B_-)\big)^{\frac{1}{n}}
\]

\[
\geq \Big(\big((mA_+)^{\frac{1}{n}} + (mB_+)^{\frac{1}{n}}\big)^{n} + \big((mA_-)^{\frac{1}{n}} + (mB_-)^{\frac{1}{n}}\big)^{n}\Big)^{\frac{1}{n}}
\]

\[
= \Big(\big((t\, mA)^{\frac{1}{n}} + (t\, mB)^{\frac{1}{n}}\big)^{n} + \big(((1-t)\, mA)^{\frac{1}{n}} + ((1-t)\, mB)^{\frac{1}{n}}\big)^{n}\Big)^{\frac{1}{n}}
\]

\[
= (mA)^{\frac{1}{n}} + (mB)^{\frac{1}{n}}. \quad \text{成立.}
\]

Step 3. 设 $A, B$ 是紧集，$\forall \varepsilon > 0$，存在有限方体并 $A_\varepsilon, B_\varepsilon$.

s.t. $A \subset A_\varepsilon \subset A + B_\varepsilon(0)$，$B \subset B_\varepsilon \subset B + B_\varepsilon(0)$.

则 $m\big((A+B) + B_{2\varepsilon}(0)\big)^{\frac{1}{n}} \geq m\big((A + B_\varepsilon(0)) + (B + B_\varepsilon(0))\big)^{\frac{1}{n}}$

\[
\geq m\big(A_\varepsilon + B_\varepsilon\big)^{\frac{1}{n}}
\]

\[
\overset{\text{Step 2}}{\geq} (mA_\varepsilon)^{\frac{1}{n}} + (mB_\varepsilon)^{\frac{1}{n}}
\]

\[
\geq (mA - \varepsilon)^{\frac{1}{n}} + (mB - \varepsilon)^{\frac{1}{n}}
\]

令 $\varepsilon \to 0+$ 即得证.

Step 4. $A$、$B$ 是一般正测集.

\[
(mA)^{\frac{1}{n}} + (mB)^{\frac{1}{n}} = \Big(\sup_{\text{紧} K_A \subset A} mK_A\Big)^{\frac{1}{n}} + \Big(\sup_{\text{紧} K_B \subset B} mK_B\Big)^{\frac{1}{n}}
\]

\[
= \sup \ \big((mK_A)^{\frac{1}{n}} + (mK_B)^{\frac{1}{n}}\big)
\]

\[
\overset{\text{Step 3}}{\leq} \sup\ \big(m(K_A + K_B)\big)^{\frac{1}{n}}
\]

\[
\leq \sup_{\text{紧} K \subset A + B} (mK)^{\frac{1}{n}} \ \leq\ m^*(A+B)^{\frac{1}{n}}
\]

综上定理得证. $\square$

\secn{6. 不可测集（以 Vitali 集为例）}

\textbf{def.}（Vitali set）

称 $A \subset \mathbb{R}$ 是 V 集，若

（1）$\forall r \neq s \in \mathbb{Q}$，$(A+r) \cap (A+s) = \emptyset$.

（2）$\mathbb{R} = A + \mathbb{Q}$

$\mathbb{R}^n$ 中的 Vitali set $V$：

（1）$\forall \vec{r} \neq \vec{s} \in \mathbb{Q}^n$，$(V + \vec{r}) \cap (V + \vec{s}) = \emptyset$

（2）$\mathbb{R}^n = V + \mathbb{Q}^n$

\textbf{Prop.} 设 Vitali set $A \subset \mathbb{R}$.

$\mathbb{R}$ 视为 $\mathbb{Q}$-线性空间，$\mathbb{R}/\mathbb{Q} = \{[x] \mid x \in \mathbb{R}\}$.

$A \cap [x]$ 是单点集，$\forall x \in \mathbb{R}$.

\textbf{Prop.} Vitali set 是 L-不可测集.

pf. 反设 Vitali Set $A$ 可测.

Case 1. $mA > 0$. 由 Steinhaus' Thm 2，$A - A$ 以 0 为内点，

$A \cap (A+r) \neq \emptyset$ 对充分小的 $r \in \mathbb{Q}$ 成立.

与定义中（1）矛盾.

Case 2. $mA = 0$. 由 $\mathbb{R} = A + \mathbb{Q}$，

\[
m\ \mathbb{R} = \sum_{r \in \mathbb{Q}} m(A+r) = \sum_{r \in \mathbb{Q}} mA = 0 \quad \text{矛盾}.
\]

故 $A$ 不可测. $\square$

\textbf{Rem.} 由 Hamel basis 到 Vitali set

Hamel basis $H$，$\mathbb{R} = \mathrm{span}_{\mathbb{Q}}(H)$，s.t.

（1）$\forall \{x_i\}_{i=1}^{k} \subset H$，$\sum_{i=1}^{k} q_i x_i = 0 \Rightarrow q_i = 0$（$q_i \in \mathbb{Q}$）

（2）$\forall x \in \mathbb{R}$，$\exists \{x_i\}_{i=1}^{k} \subset H$，$\{q_i\}_{i=1}^{k} \subset \mathbb{Q}$，s.t. $x = \sum_{i=1}^{k} q_i x_i$

其存在性由 Zorn 引理保证.

现设 $1 \in H$，则 $A := \mathrm{span}_{\mathbb{Q}}(H \setminus \{1\}) \subset \mathbb{R}$ 是 Vitali set，即 $A$ 是余维数为 1 的子集.

\textbf{Prop.} $\forall$ 正测集 $E \subset \mathbb{R}^n$，$\exists$ 不可测集 $A \subset E$.

pf. 取 Vitali 集 $V \subset \mathbb{R}^n$，$\sum\limits_{r \in \mathbb{Q}^n} m^*\big((V+r) \cap E\big) \geq m^*E > 0$

$\exists r \in \mathbb{Q}^n$，s.t. $m^*\big((V+r) \cap E\big) > 0$. 记 $A = (V+r) \cap E$，

若 $A$ 可测，则 $\exists$ 正测紧集 $K \subset A$，$mK > 0$，$\{K + r\}_{r \in \mathbb{Q}^n}$ 两两不交.

则 $H := \bigsqcup\limits_{r \in \mathbb{Q}^n} (K+r)$ 有界，$mH < \infty$. 另一方面，$mH = \sum\limits_{r \in \mathbb{Q}^n} mK = \infty$. 矛盾.

故 $A$ 不可测，即为所求. $\square$

\secn{1. 连续映射与可测映射}

\textbf{def.} 设 $X, Y$ 是拓扑空间. $f: X \to Y$. 若开集在 $f$ 下的原像是开集，则称 $f$ 连续.

\textbf{Prop.} 设 $(Y, \mathcal{Y})$ 是拓扑空间，$X$ 是非空集，$f: X \to Y$，则

\[
\mathcal{Y}_X := \{f^{-1}(T) : T \in \mathcal{Y}\}
\]

是 $X$ 上使得 $f$ 连续的最小拓扑.

pf. $\forall \{f^{-1}(T_\alpha)\} \subset \mathcal{Y}_X$，$\bigcup_\alpha f^{-1}(T_\alpha) = f^{-1}\big(\bigcup_\alpha T_\alpha\big) \in \mathcal{Y}_X$

故 $\mathcal{Y}_X$ 是拓扑.

$\forall \mathcal{P}$ on $X$，若 $f: X \to Y$ 连续，则 $f^{-1}(T) \in \mathcal{P}$，$\forall T \in \mathcal{Y}$.

故 $\mathcal{Y}_X \subset \mathcal{P}$.

\textbf{Prop.} 设 $(X, \mathcal{Y})$ 是拓扑空间，$Y \neq \emptyset$，$f: X \to Y$，则

\[
\mathcal{Y}_Y := \{T \subset Y \mid f^{-1}(T) \in \mathcal{Y}\}
\]

是 $Y$ 上使得 $f$ 连续的最大拓扑.

pf. $\forall \{T_\alpha\} \subset \mathcal{Y}_Y$，$\bigcup_\alpha f^{-1}(T_\alpha) = f^{-1}\big(\bigcup_\alpha T_\alpha\big) \in \mathcal{Y}$.

故 $\mathcal{Y}_Y$ 是拓扑.

$\forall \mathcal{P}$ on $Y$，若 $f: X \to Y$ 连续，则 $f^{-1}(T) \in \mathcal{Y}$，$\forall T \in \mathcal{P}$.

故 $\mathcal{P} \subset \mathcal{Y}_Y$.

\textbf{def.}（测度空间）设 $X \neq \emptyset$，$\mathscr{A}$ 是 $X$ 上的 $\sigma$ 代数，则称二元组 $(X, \mathscr{A})$ 为可测空间.

称 $m: \mathscr{A} \to [0, +\infty]$ 为一个测度，若

（1）$m(\emptyset) = 0$

（2）$m\Big(\bigsqcup\limits_{i=1}^{\infty} A_i\Big) = \sum\limits_{i=1}^{\infty} mA_i$ 对任一可数不交集族 $\{A_i\} \subset \mathscr{A}$ 成立.

此时称三元组 $(X, \mathscr{A}, m)$ 是测度空间.

\textbf{def.}（可测映射）设 $(X, \mathscr{A})$、$(Y, \mathcal{B})$ 是可测空间.

称 $f: X \to Y$ 是可测映射，若 $\forall B \in \mathcal{B}$，$f^{-1}(B) \in \mathscr{A}$.

\textbf{Prop.} 设 $(Y, \mathcal{B})$ 是可测空间，$X$ 是非空集，$f: X \to Y$，则

\[
\mathscr{A} := \{f^{-1}(B) : B \in \mathcal{B}\}
\]

是 $X$ 上使得 $f$ 可测的最小 $\sigma$ 代数.

\textbf{Prop.} 设 $(X, \mathscr{A})$ 是可测空间，$Y \neq \emptyset$，$f: X \to Y$，则

\[
\mathcal{B} := \{B \subset Y \mid f^{-1}(B) \in \mathscr{A}\}
\]

是 $Y$ 上使得 $f$ 可测的最大 $\sigma$ 代数.

\textbf{Prop.} 设 $\mathcal{B}$ 是 $Y$ 上的 Borel $\sigma$ 代数，若 $f: X \to Y$ 连续，

则 $f^{-1}(B)$ 是 $X$ 上的 Borel 集，$\forall B \in \mathcal{B}$.

pf. 构造 $\mathcal{B}' = \{E \subset Y : f^{-1}(E)$ 是 Borel 集$\}$，则 $\mathcal{B}'$ 是 $Y$ 上的 $\sigma$ 代数. 又开集 $\in \mathcal{B}'$，故 $\mathcal{B}'$ 包含 $Y$ 中的所有开集，即

\[
\mathcal{B} \subset \mathcal{B}'. \qquad \square
\]

\textbf{Claim.} 连续映射一定是 Borel 可测映射.

\textbf{def.} 称定义实值函数 $f: \mathbb{R}^n \to \mathbb{R}^m$ 是 Lebesgue 可测函数，若 $\forall$ Borel 集 $B \subset \mathbb{R}^m$，$f^{-1}(B)$ 是 Lebesgue 可测的.

\textbf{e.g.} 存在连续 $f$，$f($某 0 测集$) =$ 正测集.

进而存在连续 $f$，$f($某 0 测集$) =$ 不可测集.

\textbf{def.} 称定义实值函数 $f: E \to \overline{\mathbb{R}}$ 是可测函数，若 $\forall t \in \mathbb{R}$，

$\{f > t\} := \{x \in E \mid f(x) > t\}$（称为上水平集）总是可测的.

\textbf{Prop.} $f: E \to \overline{\mathbb{R}}$ 可测

$\iff \forall t \in \mathbb{R}$，$\{f \geq t\}$ 可测

$\iff \forall t \in \mathbb{R}$，$\{f < t\}$ 可测

$\iff \forall t \in \mathbb{R}$，$\{f \leq t\}$ 可测

pf. $\{f \geq t\} = \bigcap\limits_{k=1}^{\infty} \{f > t - \tfrac{1}{k}\}$

$\{f > t\} = \bigcup\limits_{k=1}^{\infty} \{f \geq t + \tfrac{1}{k}\}$

其余取补集即可. $\square$

\textbf{Prop.} 连续函数是可测函数.

\textbf{Prop.} 几乎处处连续函数是可测函数.

\textbf{Prop.} 设 $f$ 可测，则 $f^2$ 可测.

pf. $\{f^2 > t\} = \begin{cases} \mathrm{Dom} f, & \text{if } t < 0 \\ \{f > \sqrt{t}\} \cup \{f < -\sqrt{t}\}, & \text{if } t \geq 0 \end{cases}$.

故 $f$ 可测 $\Rightarrow f^2$ 可测.

\textbf{Prop.} 设 $f, g$ 可测，则 $f \pm g$，$f \cdot g$ 可测.

pf. $\{f - g > t\} = \bigcup\limits_{r \in \mathbb{Q}} \big(\{f > r\} \cap \{g < r - t\}\big)$

故 $f, g$ 可测 $\Rightarrow f \pm g$ 可测.

$f \cdot g = \tfrac{1}{4}\big((f+g)^2 - (f-g)^2\big)$ 可测. $\square$

\textbf{Prop.} 设 $f$ 可测，则 $f_+(x) := \max\{f, 0\}$，$f_-(x) := \max\{-f, 0\}$ 可测.

pf. $\{f_+ > t\} = \begin{cases} \mathrm{Dom} f, & \text{if } t < 0 \\ \{f > t\}, & \text{if } t \geq 0 \end{cases}$.

故 $f$ 可测 $\Rightarrow f_+$ 可测. 同理 $f_-$ 可测.

进而 $\vert f\vert = f_+ + f_-$ 可测. $\square$

\textbf{Prop.} 设 $f_k$ 可测，则 $\sup\limits_k f_k(x)$，$\inf\limits_k f_k(x)$ 可测.

进而 $\overline{\lim} f_k = \inf\limits_{j \geq 1} \sup\limits_{k \geq j} f_k(x)$，$\underline{\lim} f_k = \sup\limits_{j \geq 1} \inf\limits_{k \geq j} f_k(x)$ 可测.

pf. $\{\sup\limits_k f_k > t\} = \bigcup\limits_{k=1}^{\infty} \{f_k > t\}$. $\square$

\textbf{Prop.} 若 $f_k: E \to \overline{\mathbb{R}}$ 可测，逐点收敛到 $f$，则 $f$ 可测.

pf. $f = \lim f_k = \overline{\lim} f_k = \underline{\lim} f_k$. $\square$

\textbf{def.}（simple function）

称 $f: E \to \mathbb{R}$ 是简单函数，若 $f(E)$ 有限.

即存在 $E$ 的有限划分 $E = \bigsqcup\limits_{i=1}^{k} E_i$，$f(E_i) = \{y_i\} \subset \mathbb{R}$.

\textbf{def.}（简单可测函数）$f$ 是简单函数 且 $f^{-1}(y)$ 可测，$\forall y \in \mathbb{R}$.

\textbf{Thm.}（简单函数逼近可测函数）

设 $f: E \to [0, \infty]$ 非负可测，则存在非负简单可测函数列 $\{\varphi_k\}$，

$\varphi_k \leq \varphi_{k+1}$ 且 $\varphi_k \to f$  pointwise.

pf. 容易构造

\[
\varphi_k = \sum_{j=0}^{2^{2k}-1} \frac{j}{2^k} \cdot 1_{\{\frac{j}{2^k} \leq f < \frac{j+1}{2^k}\}} + 2^k \cdot 1_{\{f \geq 2^k\}}
\]

其中 $1_A$ 是示性函数（$1_A$ 是可测函数 $\iff A$ 是可测集）. $\square$

\textbf{Thm.} 设 $(E, \mathscr{A})$ 是可测空间，则 $\forall f: E \to \overline{\mathbb{R}}\ \ \mathscr{A}$ 可测，$\exists$ 简单 $\mathscr{A}$ 可测函数列 $\varphi_k \to f$  pointwise.

\secn{2. 可测函数收敛定理}

\textbf{Def.} 设 $f, g: E \to \overline{\mathbb{R}}$，若 $m\{f \neq g\} = 0$，则称 $f$ 与 $g$ 几乎处处相等：$f(x) = g(x)$，a.e. $x \in E$.

\textbf{Prop.} 设 $f, g: E \to \overline{\mathbb{R}}$，$f$ 可测，$g = f$ a.e.，则 $g$ 可测.

pf. $\{g > t\} = \{f > t\} \cup \{g \neq f,\ g > t\}$. $\square$

\textbf{Def.}（几乎处处收敛）称可测函数列 $f_k$ 几乎处处收敛到 $f$，

若 $m\{\lim\limits_k f_k \neq f\} = 0$.

\textbf{Def.} 称 $f: E \to \overline{\mathbb{R}}$ Borel 可测，若 $\{f > t\}$ 是 Borel 集.

**Prop(*)** 若 $f: E \to \overline{\mathbb{R}}$ L 可测，则 $\exists\, g$ Borel 可测，$f = g$ a.e.

pf. Step 1. 示性函数 $1_A$ $\begin{cases} \text{L 可测} \iff A \text{ L可测} \\ \text{B 可测} \iff A \text{ 是 Borel 集} \end{cases}$

设 $A$ 是 L 可测集，$f = 1_A$ 是 L 可测函数.

$\exists$ Borel 集 $B$，$m(A \triangle B) = 0$

令 $g = 1_B$ 是 B 可测函数，$m\{f \neq g\} = m(A \triangle B) = 0$

$\Rightarrow f = g$ a.e.

Step 2. 简单函数 $\varphi = \sum\limits_{i=1}^{n} c_i 1_{A_i}: E \to \mathbb{R}$

令 $\psi = \sum\limits_{i=1}^{n} c_i 1_{B_i}$，其中 $B_i$ 是 Borel 集且 $m(A_i \triangle B_i) = 0$.

则 $\varphi = \psi$，a.e. $x \in E$.

Step 3. 一般可测函数 $f = \lim\limits_{k \to \infty} \varphi_k$  pointwise，

$\varphi_k$ 简单可测 $\Rightarrow \exists$ Borel 简单可测 $\psi_k: E \to \mathbb{R}$，

$\psi_k = \varphi_k$ a.e. $x \in E$. 令 $g = \lim\limits_k \psi_k$  $(\ast)$

$(\ast): \forall k$，$\psi_k = \varphi_k$ a.e. $\Rightarrow m \bigcup_k \{\psi_k \neq \varphi_k\} = 0$

则 $\forall x \in \bigcap\limits_{k=1}^{\infty} \{\psi_k = \varphi_k\}$，$f(x) = \lim\limits_k \varphi_k(x) = \lim\limits_k \psi_k(x)$

即 $\psi_k$ 几乎处处收敛到 $f$.

$\psi_k$ Borel 可测 $\Rightarrow g = \lim \psi_k$ 存在且 Borel 可测.

则 $g = f$，a.e. $x \in E$. $\square$

\textbf{Def.}（依测度收敛）

设 $f_k: E \to \overline{\mathbb{R}}$ 是几乎处处有限的可测函数列，$f$ 可测，

称 $f_k \to f$ in measure，若 $\forall \varepsilon > 0$，

\[
\lim_k\ m\{\vert f_k - f\vert > \varepsilon\} = 0
\]

\textbf{Claim.} 上述 $f$ 几乎处处有限.

\[
\{f = \pm\infty\} \subset \bigcup \{f_k = \pm\infty\} \cup \{f_k \in \mathbb{R},\ \lim f_k = \pm\infty\} =: G
\]

$mG = m\{x \in G : \vert f_k - f\vert > \varepsilon\} \to 0$（$k \to \infty$）

而 $mG$ 与 $k$ 无关，则 $mG = 0$. 从而 $m\{f = \pm\infty\} = 0$. $\square$

\textbf{Prop.}（Lebesgue）设 $mE < \infty$，几乎处处有限可测 $f_k \to f$，a.e.

则 $f_k \to f$ in measure.

pf. 记 $E_k(\varepsilon) = \{x \in E \mid \vert f_k(x) - f(x)\vert > \varepsilon\}$

只需 $\forall \varepsilon > 0$，$\lim\limits_k mE_k(\varepsilon) = 0$.

Claim. $f_k \to f$ a.e. $\Rightarrow m\ \overline{\lim}\ E_k(\varepsilon) = 0$ 这是因为：

$\forall x \in \overline{\lim}\ E_k(\varepsilon)$，$\exists \{k_i\}$，s.t. $x \in E_{k_i}(\varepsilon)$，则 $\vert f_{k_i}(x) - f(x)\vert > \varepsilon$.

于是 $\overline{\lim}\ E_k(\varepsilon) \subset \{x \mid \lim f_k(x) \neq f(x)\}$ 而右侧零测.

从而 $m \bigcap\limits_{j=1}^{\infty} \bigcup\limits_{k \geq j} E_k(\varepsilon) = 0$，且 $\bigcup\limits_{k \geq j} E_k(\varepsilon)$ 关于 $j$ 递减.

又 $mE < +\infty$，故 $\lim\limits_{j \to \infty} m \bigcup\limits_{k \geq j} E_k(\varepsilon) = 0$.

即 $\lim\limits_{k \to \infty} mE_k(\varepsilon) = 0$，于是 $f_k \to f$ in measure. $\square$

\textbf{Def.}（近一致收敛）

称 $f_k$ 近一致收敛到 $f$，若 $\forall \varepsilon > 0$，$\exists E^\varepsilon \subset E$，$m(E \setminus E^\varepsilon) < \varepsilon$

s.t. $f_k\big\vert_{E^\varepsilon} \Rightarrow f\big\vert_{E^\varepsilon}$，$k \to \infty$.

\textbf{Thm.}（Egoroff）设 $mE < \infty$，若 $f_k \to f$，a.e. $x \in E$，则 $f_k$ 近一致收敛于 $f$.

pf. $\forall \tfrac{1}{2^j}$，$\exists k_j$，s.t. $m\Big(\bigcup\limits_{k \geq k_j} E_k\big(\tfrac{1}{2^j}\big)\Big) < \dfrac{\varepsilon}{2^j}$

\[
m\Big(\bigcup_{j \geq 1} \bigcup_{k \geq k_j} E_k\big(\tfrac{1}{2^j}\big)\Big) \ \leq\ \sum_{j \geq 1} m\Big(\bigcup_{k \geq k_j} E_k\big(\tfrac{1}{2^j}\big)\Big) < \varepsilon.
\]

令 $E^\varepsilon = \Big(\bigcup\limits_{j \geq 1} \bigcup\limits_{k \geq k_j} E_k\big(\tfrac{1}{2^j}\big)\Big)^c = \bigcap\limits_{j \geq 1} \bigcap\limits_{k \geq k_j} \Big\{\vert f_k - f\vert \leq \tfrac{1}{2^j}\Big\}$

则 $\forall \tfrac{1}{2^j}$，$\exists k_j$，s.t. $\forall k \geq k_j$，$\forall x \in E^\varepsilon$，$\vert f_k(x) - f(x)\vert \leq \tfrac{1}{2^j}$.

即 $f_k$ 在 $E^\varepsilon$ 上一致收敛到 $f$. $\square$

\textbf{Thm.} 若几乎处处有限 $f_k$ 近一致收敛于 $f$，则

（1）$f_k \to f$，a.e.  （2）$f_k \to f$ in measure.

pf.（1）$\forall \tfrac{1}{2^j}$，$\exists E^j \subset E$，$m(E \setminus E^j) < \tfrac{1}{2^j}$. 令 $\widehat{E} = \bigcup E^j$，

则 $m(E \setminus \widehat{E}) = 0$. 又易知 $\forall x \in \widehat{E}$，$\exists E^j$ s.t. $x \in E^j$，

$\lim f_k(x) = f(x)$ for $x \in E^j$. 故 $f_k \to f$，a.e.

（2）$\forall \varepsilon > 0$，$\forall \tfrac{1}{2^j}$，$\exists E^j \subset E$，$m(E \setminus E^j) < \tfrac{1}{2^j}$. 并且

$\vert f_k(x) - f(x)\vert \leq \dfrac{\varepsilon}{2}$，$\forall x \in E^j$，当 $k$ 充分大时.

即 $\{\vert f_k - f\vert > \varepsilon\} \subset E \setminus E^j$，从而

$m\{\vert f_k - f\vert > \varepsilon\} < \dfrac{1}{2^j} \quad \Rightarrow \quad \lim\limits_k m\{\vert f_k - f\vert > \varepsilon\} = 0.$ $\square$

\textbf{Def.}（依测度 Cauchy 列）

称几乎处处有限可测函数列 $f_k$ 为依测度 Cauchy 列，

若 $\forall \varepsilon > 0$，$\lim\limits_{n, m \to \infty} m\{\vert f_m - f_n\vert > \varepsilon\} = 0$.

\textbf{Thm.} 设 $\{f_k\}$ 是 $E$ 上几乎处处有界的可测函数，则

$f_k$ 依测度收敛 $\iff f_k$ 是 Cauchy 列 in measure.

pf. $\Rightarrow$. $\{\vert f_n - f_m\vert > \varepsilon\} \subset \{\vert f_n - f\vert > \tfrac{\varepsilon}{2}\} \cup \{\vert f_m - f\vert > \tfrac{\varepsilon}{2}\}$

$mL \leq mR_1 + mR_2 \to 0$，$n, m \to \infty$.

$\Leftarrow$. 对 $\forall i \in \mathbb{N}$，$\exists k_i$，s.t. $\forall \ell, j \geq k_i$，有

\[
m\{x \in E : \vert f_\ell(x) - f_j(x)\vert \geq \tfrac{1}{2^i}\} < \tfrac{1}{2^i}
\]

设 $k_i < k_{i+1}$. 令 $E_i := \{x \in E : \vert f_{k_i}(x) - f_{k_{i+1}}(x)\vert \geq \tfrac{1}{2^i}\}$.

则 $mE_i < 2^{-i}$. 令 $S = \overline{\lim}\ E_i = \bigcap\limits_{j=1}^{\infty} \bigcup\limits_{i=j}^{\infty} E_i$，则 $mS = 0$.

若 $x \notin S$，则 $\exists \overline{j}$，s.t. $x \in E \setminus \bigcup\limits_{i=\overline{j}}^{\infty} E_i$，从而当 $i \geq \overline{j}$ 时，

\[
\vert f_{k_{i+1}}(x) - f_{k_i}(x)\vert < 2^{-i}
\]

则 $l \geq \overline{j}$ 时，有 $\sum\limits_{i=l}^{\infty} \vert f_{k_{i+1}}(x) - f_{k_i}(x)\vert \leq \dfrac{1}{2^{l-1}}$.

故 $f_{k_l}(x) + \sum\limits_{i=1}^{\infty} \big[f_{k_{i+1}}(x) - f_{k_i}(x)\big]$ 在 $E \setminus S$ 上绝对收敛. 从而

$\{f_{k_i}(x)\}$ 在 $E$ 上几乎处处收敛，设收敛于 $f(x)$，$f(x)$ 是 $E$ 上几乎处处有限的可测函数.

而 $f_{k_i}(x)$ 在 $E$ 上近一致收敛 $\Rightarrow$ 依测度收敛.

则 $m\{\vert f_k(x) - f(x)\vert \geq \varepsilon\} \leq m\{\vert f_k - f_{k_l}\vert \geq \tfrac{\varepsilon}{2}\} + m\{\vert f_{k_l} - f\vert \geq \tfrac{\varepsilon}{2}\}$

$k \to \infty$ 时 RHS $\to 0$. 从而 $f_k \to f$ in measure. $\square$

\textbf{Thm.}（Riesz）设 $f_k \to f$ in measure，则 $f_k$ 存在子列 $f_{k_i}$ 几乎处处收敛于 $f$.

pf. 由前述关于 Cauchy 列的证明，可知

$\exists$ 子列 $f_{k_i}$ 近一致收敛到 $\widetilde{f}$.

则 $f_{k_i} \to \widetilde{f}$ a.e. 且 in measure.

由极限函数唯一性有：$\widetilde{f} = f$ a.e.

故 $f_{k_i} \to f$ a.e. $\square$

Summary.（各收敛之间的关系图）

\begin{itemize}

\item 几乎处处收敛 $\longrightarrow$ 依测度收敛（$(\ast)\ mE < \infty$，Lebesgue）

\item 依测度收敛 $\longrightarrow$ 几乎处处收敛（子列，Riesz）

\item 几乎处处收敛 $\longrightarrow$ 近一致收敛（$mE < \infty$，Egoroff）

\item 近一致收敛 $\longrightarrow$ 几乎处处收敛

\item 近一致收敛 $\longrightarrow$ 依测度收敛

\item 依测度收敛 $\longrightarrow$ 近一致收敛（子列，Cauchy）

\end{itemize}

$(\ast)$ e.g. $1_{(n, n+1)}: \mathbb{R} \to \mathbb{R}$

$1_{(n, n+1)} \to 0$，a.e.（$n \to \infty$）

但 $1_{(n, n+1)}$ 不依测度收敛.

\secn{3. 可测函数与连续函数的关系}

\textbf{Thm.}（Lusin）

设 $f$ 是 $E$ 上几乎处处有限的可测函数，则 $\forall \varepsilon > 0$，$\exists$ 闭集 $F_\varepsilon \subset E$，s.t. $m(E \setminus F_\varepsilon) < \varepsilon$，且

\[
f\vert_{F_\varepsilon} \ \text{是连续函数}.
\]

pf. Step 1. 若 $f$ 是简单函数，即 $f = \sum\limits_{k=1}^{K} c_k \cdot 1_{A_k}$，其中 $A_k$ 是两两不交的可测集列. 存在闭子集 $F_k \subset A_k$，$m(A_k \setminus F_k) < \dfrac{\varepsilon}{2^k}$ 且 $f\vert_{F_k} = c_k$ 连续. 令 $F = F_1 \cup \cdots \cup F_K$，$f\vert_F$ 连续.

Step 2. $f$ 有界可测. 则 $\exists$ 简单可测函数列 $\varphi_k \Rightarrow f$.

对每个 $\varphi_k$ 存在闭 $F_k$ 满足 $\varphi_k\vert_{F_k}$ 连续，取 $F = \bigcap\limits_{k=1}^{\infty} F_k$ 即可.

Step 3. $f$ 几乎处处有界，可测. 令 $g(x) = \dfrac{f(x)}{1 + \vert f(x)\vert}$，则

$\vert g(x)\vert < 1$ 且可测. 取闭 $F \subset E$ s.t. $g\vert_F$ 连续则 $f\vert_F$ 连续.$\square$

\textbf{Prop.} 设 $f$ 几乎处处有界且可测，则 $\forall \varepsilon > 0$，$\exists$ 连续函数

$g_\varepsilon: \mathbb{R}^n \to \mathbb{R}$，s.t. $m\{x \in E : f(x) \neq g_\varepsilon(x)\} < \varepsilon$.

pf. 由 Lusin Thm，$\exists F_\varepsilon \subset E$，s.t. $m(E \setminus F_\varepsilon) < \varepsilon$，$f\vert_{F_\varepsilon}$ 连续.

再由 Tietz 连续延拓，$\exists$ 连续 $g_\varepsilon: \mathbb{R}^n \to \mathbb{R}$，$g_\varepsilon\vert_{F_\varepsilon} = f\vert_{F_\varepsilon}$

则 $m\{f \neq g_\varepsilon\} \leq m(E \setminus F_\varepsilon) < \varepsilon$. $\square$

\textbf{Prop.} 设 $f$ 几乎处处有界，可测，则存在 $\{f_k\}$ 连续，

$f_k \to f$  a.e.

pf. 存在闭集 $F_k$，$m(E \setminus F_k) < \dfrac{\varepsilon}{2^k}$，s.t.

$\exists$ 连续函数 $f_k$，s.t. $f_k\vert_{F_k} = f\vert_{F_k}$.

取 $E_j = \bigcup\limits_{k \geq j} F_k$，$f_k\vert_{E_j} = f\vert_{E_j}$，$\forall k \geq j$.

\[
m(E \setminus E_j) < \sum_{k=j}^{\infty} m(E \setminus F_k) < \frac{\varepsilon}{2^{j-1}}.
\]

令 $\widetilde{E} = \bigcup\limits_{j \geq 1} E_j \subset E$，则 $m(E \setminus \widetilde{E}) \leq m(E \setminus E_j) < \dfrac{\varepsilon}{2^{j-1}}$，$\forall j$.

$\Rightarrow m(E \setminus \widetilde{E}) = 0$.

$\forall x \in \widetilde{E}$，$\exists j \geq 1$，$x \in E_j$，$\forall k \geq j$，$f_k(x) = f(x)$.

$\Rightarrow f_k \to f$，a.e. $x \in E$. $\square$

\secn{4. Littlewood's Principle}

\textbf{Principle 1.}（Lusin）以下叙述等价：

 \textcircled{1}  $f$ 几乎处处有限，可测

 \textcircled{2}  $\forall \varepsilon > 0$，$\exists$ 闭子集 $F \subset E$，$m(E \setminus F) < \varepsilon$ 且 $f\vert_F$ 连续.

 \textcircled{3}  $\forall \varepsilon > 0$，$\exists \widehat{E} \subset E$，$m^*(E \setminus \widehat{E}) < \varepsilon$ 且 $f\vert_{\widehat{E}}$ 连续.

 \textcircled{4}  $\forall \varepsilon > 0$，$\exists$ 连续 $g: E \to \mathbb{R}$，s.t. $m^*\{f \neq g\} < \varepsilon$.

（旁注： \textcircled{1}  \textcircled{2}  \textcircled{3}  \textcircled{4}  推出关系小图，$(\ast)$）

pf.  \textcircled{1}  $\Rightarrow$  \textcircled{2}  即普通的 Lusin Thm.

 \textcircled{2}  $\Rightarrow$  \textcircled{4}  用 Tietz 连续延拓.

 \textcircled{2}  $\Rightarrow$  \textcircled{3}  显然.

 \textcircled{3}  $\Rightarrow$  \textcircled{2}  欲从 $\widehat{E}$ 来找到闭子集 $F$.

$\forall \varepsilon > 0$，$\exists \widehat{E} \subset E$，$m^*(E \setminus \widehat{E}) < \dfrac{\varepsilon}{3}$. 注意到 $m^*$ 外正则.

故 $\exists$ 开集 $G \supset E \setminus \widehat{E}$，s.t. $mG < \dfrac{2}{3}\varepsilon$.

注意可测 $(E \cap G^c) \subset \widehat{E}$，

由 $m$ 内正则性，$\exists$ 闭集 $F \subset E \cap G^c$，s.t. $m(E \cap G^c \setminus F) < \dfrac{\varepsilon}{3}$.

另一方面 $m\big(E \setminus (E \cap G^c)\big) = m(E \cap G) \leq mG < \dfrac{2}{3}\varepsilon$.

于是 $m(E \setminus F) = m(E \cap G \setminus F) + m(E \cap G^c \setminus F) < \varepsilon$.

 \textcircled{4}  $\Rightarrow$  \textcircled{3}  取 $\widehat{E} = \{x : f(x) = g(x)\}$ 即可.

 \textcircled{2}  $\Rightarrow$  \textcircled{1}  $\forall k$，$\exists$ 闭集 $F_k \subset E$，$m(E \setminus F_k) < \dfrac{1}{k}$ 及 $f\vert_{F_k}$ 连续.

令 $F = \bigcup F_k$，则 $(E \setminus F) \subset (E \setminus F_k) \Rightarrow m(E \setminus F) = 0$.

于是 $\{x \in E : f(x) > t\} = \big(\bigcup \{x \in F_k : f(x) > t\}\big) \cup \Sigma$ 是可测集.

则 $f$ 可测. 又 $\forall x \in F$，$x \in F_k$，$f\vert_{F_k}$ 连续 $\Rightarrow f\vert_F$ 有限.

故 $f$ 是几乎处处有限的可测函数.

综合以上即已证毕. $\square$

\textbf{Principle 2.}（Egoroff）$mE < \infty$，可测 $f_k \to f$ a.e. $\iff f_k \to f$ 近一致.

\textbf{Principle 3.} $E$ 可测 且 $mE < \infty$

$\iff \forall \varepsilon > 0$，存在有限个方体 $\{Q_i\}_{i=1}^{n}$，s.t. $m^*\Big(E \triangle \big(\bigcup\limits_{i=1}^{n} Q_i\big)\Big) < \varepsilon$.

pf. $\Rightarrow$. $\exists$ 方体簇 $\{Q_n\}_{n=1}^{\infty}$，s.t. $E \subset \bigcup\limits_{n=1}^{\infty} Q_n$.

由 $\sum\limits_{j=1}^{\infty} m(Q_j) < mE + \varepsilon < \infty$，可知 $\sum\limits_{j=n+1}^{\infty} mQ_j < \varepsilon$.

于是 $m\Big(E \triangle \bigcup\limits_{i=1}^{n} Q_i\Big) \leq m\Big(E \triangle \bigcup\limits_{i=1}^{\infty} Q_i\Big) + m\Big(\bigcup\limits_{i=1}^{\infty} Q_i \triangle \bigcup\limits_{i=1}^{n} Q_i\Big) < 2\varepsilon$.

$\Leftarrow$. $\forall \tfrac{1}{2^k}$，$\exists$ 有限方体 $Q_{k1}, Q_{k2}, \cdots, Q_{k n_k}$，s.t.

\[
m^*\Big(E \triangle \bigcup_{i=1}^{n_k} Q_{k_i}\Big) =: m^*(E \triangle A_k) < \frac{1}{2^k}.
\]

则 $m^*E \leq m^*(E \triangle A_k) + m^*A_k < \dfrac{1}{2^k} + m^*A_k < \infty$.

令 $A = \underline{\lim}\ A_k = \bigcup\limits_{k \geq 1} \bigcap\limits_{j \geq k} A_j \subset E$，

\[
0 \leq m^*(E \setminus A) \ =\ m^*\Big(E \setminus \bigcap_{j \geq k} A_j\Big)
\]

\[
\leq \sum_{j \geq k} m^*(E \setminus A_j) \ \leq\ \frac{1}{2^{k-1}} \ \to\ 0
\]

则 $m^*(E \setminus A) = 0$，故 $E$ 可测. $\square$

\textbf{Principle 4.} 可测 $f: E \to \overline{\mathbb{R}}$ 几乎处处有界 $\iff$ 近似有界，即

$\forall \varepsilon > 0$，$\exists F \subset E$，$\exists M > 0$，$m(E \setminus F) < \varepsilon$，$\vert f\vert_F(x)\vert < M$.

\secn{5. 可测函数的复合}

\textbf{Prop.} 设 $f: E \to \mathbb{R}$ 实可测，$g: \mathbb{R} \to \mathbb{R}$ 连续，则 $g \circ f$ 可测.

pf. $\forall$ 开集 $G$，$(g \circ f)^{-1}(G) = f^{-1}\big(g^{-1}(G)\big)$

而 $g$ 连续 $\Rightarrow g^{-1}(G)$ 是开集，故 $g \circ f$ 可测. $\square$

\textbf{Prop.} 设 $g: \mathbb{R}^m \to \mathbb{R}^n$ 连续，$f: \mathbb{R}^n \to \mathbb{R}$ 实可测，且任一零测集在 $g$ 下的原像集为零测集. 则 $f \circ g$ 可测.

pf. $\forall$ 开集 $G$，

\[
(f \circ g)^{-1}(G) = g^{-1}\big(f^{-1}(G)\big) = g^{-1}\big(F_\sigma \cup Z\big) = g^{-1}(F_\sigma) \cup g^{-1}(Z).
\]

而 $F_\sigma$ 集在连续函数下的原像是 $F_\sigma$ 集，故可知 $f \circ g$ 可测. $\square$

\chapline{Chapter 4. Lebesgue 积分}

\secn{1. 非负可测函数的积分}

\textbf{def} 非负可测函数空间 $L^+ = \{$可测 $f: E \to [0, +\infty]\}$.

对于 simple function $\varphi \in L^+$，$\varphi(x) := \sum\limits_{j=1}^{k} a_j \cdot 1_{A_j}$，$A_j$ 两两不交可测.

定义 $\int_E \varphi \, dm := \sum\limits_{j=1}^{k} a_j\, m(A_j)$.

\textbf{Prop}（1）$\forall c \geq 0$，$\int_E c\varphi \, dm = c \cdot \int_E \varphi \, dm$.

（2）$\int_E (\varphi + \psi) \, dm = \int_E \varphi \, dm + \int_E \psi \, dm$.

（3）$\varphi \leq \psi \Rightarrow \int_E \varphi \, dm \leq \int_E \psi \, dm$.

\textbf{def.} 对 $\forall f \in L^+$，定义 $f$ 的 Lebesgue 积分

\[
\int_E f \, dm = \sup \Big\{ \int_E \varphi \, dm : 0 \leq \varphi \leq f,\ \varphi\ \text{simple} \Big\}
\]

称 $f$ 是 L 可积的，若 $\int_E f \, dm < +\infty$.

\textbf{Prop} 设 $f$ 非负可测，若 $F$ 是 L 可积的，且 $f \leq F$，则 $f$ 也是 L 可积的.

\textbf{Prop.} 设 $f$ 非负可积，则 $f(x) < \infty$  a.e. $x \in E$.

pf. 取 $E_k = \{x \in E : f(x) > k\}$，则

\[
\{x \in E : f(x) = +\infty\} = \bigcap_{k=1}^{\infty} E_k = \lim E_k.
\]

$\forall k$，$k \cdot mE_k \leq \int_{E_k} f(x)\, dm \leq \int_E f(x)\, dm < +\infty$

$\Rightarrow \lim mE_k = 0 \quad \Rightarrow \quad m(\lim E_k) = 0.$ $\square$

\textbf{Thm.}（Levi Monotone Convergence，MCT）

suppose $\{f_j\} \subset L^+$，$f \in L^+$，$f_j \leq f_{j+1}$，$f_j \to f$  pointwise.

then $\displaystyle\int_E f \, dm = \lim_{j \to \infty} \int_E f_j \, dm$.

pf. $f_j \leq f_{j+1} \Rightarrow \int_E f_j \, dm \leq \int_E f_{j+1} \, dm$. $\int_E f_j \, dm \leq \int_E f \, dm$.

$\displaystyle\lim_{j \to \infty} \int_E f_j \, dm = \sup_{j \geq 1} \int_E f_j \, dm \leq \int_E f \, dm$.

Fixed simple $0 \leq \varphi \leq f$. $\alpha \in (0, 1)$. Note $E_j := \{x \in E : f_j \geq \alpha \varphi\}$.

Then $\bigcup E_j = E$ and $E_j \subset E_{j+1}$.

\[
\lim_{j \to \infty} \int_E f_j \, dm \ \geq\ \lim_{j \to \infty} \int_{E_j} f_j \, dm \ \geq\ \lim_{j \to \infty} \int_{E_j} \alpha \varphi \, dm
\]

Assume $\varphi = \sum\limits_{i=1}^{k} a_i 1_{A_i}$，$\alpha \int_{E_j} \varphi \, dm = \alpha \sum\limits_{i=1}^{k} a_i\, m(A_i \cap E_j) \to \alpha \sum\limits_{i=1}^{k} a_i mA_i$.

Taking $j \to \infty$，$\displaystyle\lim_{j \to \infty} \int_E f_j \, dm \geq \int_E \alpha \varphi \, dm$

\[
\Rightarrow \lim_{j \to \infty} \int_E f_j \, dm \geq \int_E \varphi \, dm \quad \Rightarrow \quad \lim_{j \to \infty} \int_E f_j \, dm \geq \sup_{0 \leq \varphi \leq f} \int_E \varphi \, dm
\]

Thus $\displaystyle\lim_{j \to \infty} \int_E f_j \, dm = \int_E f \, dm$. $\square$

\textbf{Rem.} 定义下方图像 $Gf := \{(x, t) \in E \times [0, \infty] : 0 \leq t \leq f(x)\}$

$f_j \leq f_{j+1} \Rightarrow Gf_j \subset Gf_{j+1}$ 单增可测集列.

$Gf := \bigcup\limits_{j=1}^{\infty} Gf_j$，$\lim\limits_{j \to \infty} m(Gf_j) = m(Gf)$

$\Rightarrow \lim\limits_{j \to \infty} \int_E f_j \, dm = \int_E f \, dm$（并不严谨）

\textbf{Th.}（线性性）

设 $f, g$ 非负可积，$\alpha, \beta > 0$，则

\[
\int_E (\alpha f + \beta g)\, dm = \alpha \int_E f\, dm + \beta \int_E g\, dm.
\]

pf. 取简单可测 $0 \leq \varphi_j \leq f$，$\varphi_j \nearrow f$

$0 \leq \psi_j \leq g$，$\psi_j \nearrow g$

由 MCT，$\displaystyle\int_E (f+g)\, dm = \lim_{j \to \infty} \int_E (\varphi_j + \psi_j)\, dm$

\[
= \lim_{j \to \infty} \Big( \int_E \varphi_j \, dm + \int_E \psi_j \, dm \Big) = \int_E f\, dm + \int_E g\, dm. \qquad \square
\]

\textbf{Th.}（逐项积分定理）

设 $h_j \geq 0$ 可测几乎处处有限. 则

\[
\int_E \sum_{j=1}^{\infty} h_j \, dm = \sum_{j=1}^{\infty} \int_E h_j \, dm
\]

pf. By induction，$\displaystyle\int_E \sum_{j=1}^{N} h_j \, dm = \sum_{j=1}^{N} \int_E h_j \, dm$.

$f_N := \sum\limits_{j=1}^{N} h_j \nearrow$，by MCT，letting $N \to +\infty$. $\square$

\textbf{Prop} 设 $f$ 非负可测，则 $\displaystyle\int_E f \, dm = 0 \iff f = 0$，a.e. $x \in E$.

pf. $\Leftarrow$. $\displaystyle\int_E f \, dm = \sup_{0 \leq \varphi \leq f} \int_E \varphi \, dm$，设 $\varphi = \sum\limits_{j=1}^{N} a_j 1_{A_j} \leq f$.

设 $E' \subset E$，$m(E \setminus E') = 0$，$f(x) = 0$，$\forall x \in E'$.

$0 \leq \sum\limits_{j=1}^{N} a_j 1_{A_j} \leq 0 \quad \Rightarrow \quad a_j = 0$ 或 $mA_j = 0$，$\forall j$.

则 $\displaystyle\int_E \varphi \, dm = 0$，$\forall 0 \leq \varphi \leq f$. 故 $\displaystyle\int_E f \, dm = 0$.

$\Rightarrow$. 设 $m\{x \in E : f(x) > 0\} > 0$，则 $\exists k > 0$，$m\{x \in E : f(x) > \tfrac{1}{k}\} > 0$.

令 $A_k = \{x \in E : f(x) > \tfrac{1}{k}\}$，$\varphi = \tfrac{1}{k} \cdot 1_{A_k}$ 非负可测，$\varphi \leq f$.

则 $\displaystyle\int_E f \, dm \geq \int_E \varphi \, dm = \tfrac{1}{k}\, mA_k > 0$. 矛盾. $\square$

\textbf{Th.}（Fatou's Lemma）

设 $f_j \geq 0$ 可测，则 $\displaystyle\int_E \underline{\lim} f_j \, dm \leq \underline{\lim} \int_E f_j \, dm$.

pf. $\displaystyle\underline{\lim} f_j(x) = \lim_{k \to \infty} \inf_{j \geq k} f_j(x)$

\[
LHS = \int_E \lim_{k \to \infty} \inf_{j \geq k} f_j \, dm \ \overset{\text{MCT}}{=}\ \lim_{k \to \infty} \int_E \inf_{j \geq k} f_j \, dm
\]

\[
= \underline{\lim} \int_E \inf_{j \geq k} f_j \, dm \ \leq\ \underline{\lim} \int_E f_k \, dm \ =\ RHS. \qquad \square
\]

\textbf{Rem.} $G_{\underline{\lim} f_j} = \underline{\lim}\ Gf_j$

$LHS = m\big(\underline{\lim}\ Gf_j\big) \leq \underline{\lim}\ m(Gf_j) = RHS$

\secn{2. 广义实值可测函数积分}

\textbf{def.} 设 $f: E \to \overline{\mathbb{R}}$ 可测，$f = f_+ - f_-$.

$f_+ := \max\{f, 0\}$，$f_- := \max\{-f, 0\}$.

若 $f_+ , f_-$ 至少有一个可积，则定义 $\displaystyle\int_E f \, dm = \int_E f_+ \, dm - \int_E f_- \, dm$.

若 $f_+$ 与 $f_-$ 均可积，则称 $f$ 可积.

\textbf{Prop} 注意到 $\vert f\vert = f_+ + f_-$，则 $\displaystyle\int_E f \, dm < \infty \iff \int_E \vert f\vert \, dm < \infty$.

\textbf{Th.}（控制收敛定理，DCT）

设 $f_j$ 可测，$g \geq 0$ 可积，$f_j \to f$  pointwise，$\vert f_j\vert \leq g$.

则 $\displaystyle\lim_{j \to \infty} \int_E f_j \, dm = \int_E f \, dm$.

pf. $f = \lim\limits_{j \to \infty} f_j$ 可测. $\vert f_j\vert \leq g \Rightarrow \vert f\vert \leq g \Rightarrow f$ 可积.

考虑 $g - f_j \geq 0$，由 Fatou's Lemma，

\[
\int_E \underline{\lim} (g - f_j) \, dm \ \leq\ \underline{\lim} \int_E (g - f_j) \, dm
\]

\[
\Rightarrow \int_E (g - f) \, dm \ \leq\ \int_E g \, dm - \overline{\lim} \int_E f_j \, dm
\]

\[
\Rightarrow \int_E f \, dm \ \geq\ \overline{\lim} \int_E f_j \, dm
\]

考虑 $g + f_j \geq 0$，由 Fatou's Lemma，

\[
\int_E f \, dm \ \leq\ \underline{\lim} \int_E f_j \, dm
\]

故 $\displaystyle\int_E f \, dm = \lim \int_E f_j \, dm$. $\square$

MCT，Fatou's Lemma 及 Lebesgue 控制收敛定理中的 pointwise 均可改为 a.e.

\textbf{e.g.}（Fatou's Lemma）

设 $f_j \geq 0$，a.e. 则 $\displaystyle\underline{\lim} \int_E f_j \, dm \geq \int_E \underline{\lim} f_j \, dm$.

pf. 令 $Z_j = \{x \in E : f_j(x) < 0\}$，$m(Z_j) = 0$.

令 $\widehat{E} = E \setminus \bigcup Z_j$. 则 $m(E \setminus \widehat{E}) = 0$.

则 $\displaystyle\underline{\lim}_{j \to \infty} \int_E f_j \, dm = \lim_{j \to \infty} \int_{\widehat{E}} f_j \, dm \ \geq\ \int_{\widehat{E}} \lim_{j \to \infty} f_j \, dm = \int_E \lim_{j \to \infty} f_j \, dm$. $\square$

\textbf{Th.}（逐项积分定理）

设 $f_j: E \to \overline{\mathbb{R}}$ 可测，$g = \sum\limits_{j=1}^{\infty} \vert f_j\vert$ 可积. 则

\[
\int_E \sum_{j=1}^{\infty} f_j \, dm = \sum_{j=1}^{\infty} \int_E f_j \, dm.
\]

pf. 部分和 $S_N = \sum\limits_{j=1}^{N} f_j$，$\displaystyle\int_E S_N \, dm = \sum\limits_{j=1}^{N} \int_E f_j \, dm$.

$\vert S_N\vert \leq \sum\limits_{j=1}^{N} \vert f_j\vert \leq g$

由 $g$ 可积知 $g(x) < \infty$  a.e. $x \in E$，

则 $\sum\limits_{j=1}^{N} \vert f_j(x)\vert$ 收敛 $\Rightarrow \sum\limits_{j=1}^{N} f_j(x)$ 收敛，即 $S_N$ 收敛 a.e.

由控制收敛定理，$\displaystyle\int_E \sum_{j=1}^{\infty} f_j \, dm = \sum_{j=1}^{\infty} \int_E f_j \, dm$. $\square$

\secn{3. DCT 的若干讨论}

\textbf{Th.}（Lebesgue's Dominated Convergence Thm. DCT）

设 $f_j$ 可测，$g \geq 0$ 可积，$f_j \to f$  a.e.，$\vert f_j\vert \leq g$.

则 $\displaystyle\lim_{j \to \infty} \int_E f_j \, dm = \int_E f \, dm$.

\textbf{e.g.} $f_k(x) = 1_{(k,\ k + \frac{1}{k})}$，$f_k \to 0$  pointwise，$\int f_k \, dm \to 0$.

但并不存在控制函数可积. 据此可知原版定理条件过强.

\textbf{Prop.}（DCT，渐进控制版）

设 $f_k$ 可测，$f_k \to f$ a.e.，且 $\exists$ 可积 $g_k$，$g$，s.t.

\[
\vert f_k\vert \leq g_k \ \text{a.e.} \quad \text{且} \quad \int_E \vert g_k - g\vert \, dm \to 0 \quad (k \to \infty)
\]

则 $f$ 可积，且 $\displaystyle\lim \int_E f_k \, dm = \int_E f \, dm$.

pf. Step 1. 设 $f_k \geq 0$，则 $f$ 非负可测

\[
0 \leq \int_E f \, dm \ \leq\ \underline{\lim} \int_E f_k \, dm \ \leq\ \underline{\lim} \int_E g_k \, dm \ \leq\ \int_E g \, dm
\]

故 $f$ 可积.

$(g - f_k)_+ \leq g_+ = g$. 又 $(g - f_k)_+ \to (g - f)_+$  a.e.

故 $\displaystyle\int_E (g - f_k)_+ \, dm \to \int_E (g - f)_+ \, dm$.

另一方面，$0 \leq f_k \leq g_k \Rightarrow g - f_k \geq g - g_k$

$(g - f_k)_- \leq (g - g_k)_- \leq \vert g - g_k\vert \quad \Rightarrow \quad \int_E (g - f_k)_- \, dm \to 0$

于是 $\displaystyle\int_E (g - f_k) \, dm = \int_E (g - f_k)_+ \, dm + \int_E (g - f_k)_- \, dm \to \int_E (g - f) \, dm$  $(\ast)$

即 $\displaystyle\int_E f_k \, dm \to \int_E f \, dm$.

Step 2. 分别对 $f_+$ 及 $f_-$ 用 Step 1 即证. $\square$

\textbf{def.} 定义 $L(E) := \{$ integrable $f: E \to \overline{\mathbb{R}}\}$

$\Vert f\Vert_1 := \int_E \vert f\vert \, dm$ 为 $f$ 的 L¹-norm.

称 $f_k$ 依 L¹-norm 收敛于 $f$，若 $\Vert f_k - f\Vert \to 0$.

\textbf{Lemma.} 设 $\Vert f_k\Vert < C$，且 $f_k \to f$ a.e.，则

\[
\lim_{k \to \infty} \big(\Vert f_k - f\Vert + \Vert f\Vert - \Vert f_k\Vert\big) = 0
\]

pf. 由 Fatou's Lemma.

$\displaystyle\int_E \vert f\vert \, dm \leq \underline{\lim} \int_E \vert f_k\vert \, dm \leq C \quad \Rightarrow \quad f$ 可积.

被积函数 $\leq 2\vert f\vert$ 作为控制函数.

由 DCT 即已证. $\square$

\textbf{Prop.} 设 $f_k, f \in L^1(E)$，$f_k \to f$ a.e.，则

\[
\Vert f_k - f\Vert \to 0 \quad \iff \quad \Vert f_k\Vert \to \Vert f\Vert
\]

pf. $\Rightarrow$. $\Big\vert \int_E \vert f_k\vert dm - \int_E \vert f\vert \, dm\Big\vert \leq \int_E \big\vert \vert f_k\vert - \vert f\vert\big\vert \, dm \ \to\ 0$.

$\Leftarrow$. $\lim \int_E \vert f_k\vert \, dm = \int_E \vert f\vert \, dm < +\infty \Rightarrow \exists C$，s.t. $\int_E \vert f_k\vert \, dm < C$.

由 Lemma，

\[
\int_E \vert f_k - f\vert \, dm = \int_E \big(\vert f - f_k\vert + \vert f\vert - \vert f_k\vert\big) dm + \int_E \big(\vert f_k\vert - \vert f\vert\big) dm
\]

（两项均 $\downarrow 0$）

即已证毕. $\square$

\textbf{Lemma} 依 L¹ 收敛 $\Rightarrow$ 依测度收敛.

pf. $\Vert g_k - g\Vert \to 0$ 即 $\displaystyle\int_E \vert g_k - g\vert \, dm \to 0$.

$\forall \delta > 0$，$\displaystyle\int_E \vert g_k - g\vert \, dm \geq \int_{\{\vert g_k - g\vert > \delta\}} \vert g_k - g\vert \, dm$

\[
\geq \delta \cdot m\{\vert g_k - g\vert > \delta\}
\]

$\Rightarrow m\{\vert g_k - g\vert > \delta\} \to 0$. $\square$

\textbf{Th}（DCT，完整版）

设 $f_k \in L(E)$，$f_k \to f$ a.e.，则以下叙述等价：

（1）$\Vert f_k - f\Vert \to 0$，$k \to \infty$

（2）$f \in L(E)$，且 $\Vert f_k\Vert \to \Vert f\Vert$

（3）$\exists g_k, g \in L(E)$，s.t. $\vert f_k\vert \leq g_k$ 且 $\Vert g_k - g\Vert \to 0$

且 (1)(2)(3) 蕴含 (4)：$f \in L(E)$，$\displaystyle\int_E f_k \, dm \to \int_E f \, dm$.

pf.（1）$\iff$（2）由 Prop 即得.

（3）$\Rightarrow$（2）在 $(\ast)$ 处补证 $\int_E (g - f)_- \, dm = 0$ 即可.

By Riesz Thm，$\exists \{g_{k_i}\} \subset \{g_k\}$，s.t. $g_{k_i} \to g$，a.e.

因 $g_{k_i} \geq f_{k_i} \geq 0$，令 $k_i \to \infty$ 有 $g \geq f \geq 0$，a.e.

则 $g - f \geq 0$，a.e. 故 $(g - f)_- = 0$ a.e. 由 Prop (3) $\Rightarrow$ (4)

于是 $\displaystyle\int_E \vert f_k\vert \, dm = \int_E (f_k)_+ \, dm + \int_E (f_k)_- \, dm \to \int_E \vert f\vert \, dm$.

（1)(2) $\Rightarrow$（3）令 $g_k = \vert f_k\vert$，$g = \vert f\vert$. 则

\[
\Vert g_k - g\Vert \leq \Vert f_k - f\Vert \to 0. \quad k \to \infty. \qquad \square
\]

\textbf{Th.}（绝对连续）

设 $f \in L^1(E)$，则 $\forall \varepsilon > 0$，$\exists \delta > 0$，s.t. $\forall A \subset E$，当 $mA < \delta$ 时，

\[
\Big\vert \int_A f \, dm \Big\vert < \varepsilon.
\]

pf. 可证 $\int_A \vert f\vert \, dm < \varepsilon$，则不妨设 $f \geq 0$.

$\exists$ 简单可测 $\varphi_k \nearrow f$，$\displaystyle\int_A \varphi_k \, dm \to \int_A f \, dm$.

取简单可测 $\varphi$，s.t. $\displaystyle\int_A (f - \varphi) \, dm < \dfrac{\varepsilon}{2}$.

设 $M = \max\limits_{x \in A} \varphi(x) < \infty$，则 $\forall A \subset E$，$mA < \delta$，

\[
\int_A \varphi \, dm \ \leq\ M \cdot \delta \ =\ \frac{\varepsilon}{2} \quad ，\ \text{于}\ \delta = \frac{\varepsilon}{2M}.
\]

则 $\displaystyle\int_A f \, dm < \int_A \varphi \, dm + \dfrac{\varepsilon}{2} = \varepsilon$. $\square$

\textbf{Thm.}（DCT，依测度版）

设 $f_k \in L(E)$，$f_k \to f$ in measure，$\exists g \in L(E)$，s.t. $\vert f_k\vert \leq g$.

则 $\displaystyle\int_E f_k \, dm \to \int_E f \, dm$.

pf. $\exists f_{k_i} \to f$ a.e.，由 Fatou's Lemma，

\[
\int_E \vert f\vert \, dm \leq \underline{\lim}_{k_i} \int_E \vert f_{k_i}\vert \, dm \leq \int_E \vert g\vert \, dm < \infty \quad \text{故}\ f \in L(E).
\]

不妨设 $f_k \geq 0$，$\forall \delta > 0$，$m\{\vert f_k - f\vert > \delta\} \to 0$. 对 $\forall \varepsilon > 0$，

\[
\int_{E \cap B_N^c} g \, dm < \varepsilon
\]

\[
\int_{E \cap B_N \cap \{\vert f_k - f\vert > \delta\}} \vert f_k - f\vert \, dm < \varepsilon \quad \text{（对}\ k\ \text{充分大，绝对连续）}
\]

\[
\int_{E \cap B_N \cap \{\vert f_k - f\vert \leq \delta\}} \vert f_k - f\vert \, dm \leq \delta \cdot m(B_N) < \varepsilon. \qquad \square
\]

另证. By Riesz Thm，$\exists$ 子列 $f_{k_i} \to f$ a.e.，则 $f \in L(E)$ 且 $\Vert f_{k_i} - f\Vert \to 0$.

下证 $\overline{\lim} \int_E \vert f_k\vert \, dm \leq \int_E \vert f\vert \, dm \leq \underline{\lim} \int_E \vert f_k\vert \, dm$.

反设 $\overline{\lim} \int_E \vert f_k\vert \, dm > \int_E \vert f\vert \, dm + 2\delta$.

则 $\exists f_{k_j}$，s.t. $\displaystyle\int_E \vert f_{k_j}\vert \, dm > \int_E \vert f\vert \, dm + \delta$.

由 Riesz Thm，$\exists f_{k_{j_i}} \to f$ a.e. 从而与上式矛盾.

另一边同理. 从而立即有 $\displaystyle\int_E \vert f\vert \, dm = \lim \int_E \vert f_k\vert \, dm$. $\square$

\textbf{Prop} 设 $f \in L(E)$，则 $\forall \varepsilon > 0$，$\exists$ 有界可测 $B$，s.t. $\displaystyle\int_{E \setminus B} \vert f\vert \, dm < \varepsilon$.

pf. 取 $f \cdot 1_{E \cap B_k} \nearrow f$（$k \to \infty$），由 MCT 即有

\[
\lim_{k \to \infty} \int_E f \cdot 1_{E \cap B_k} \, dm = \int_E f \, dm < \infty
\]

从而 $\displaystyle\lim_{k \to \infty} \int_{E \setminus B_k} f \, dm = 0$ 即可得证. $\square$

\textbf{Prop} 设 $f: E \times (a, b) \to \mathbb{R}$ 关于 $x \mapsto f(x, y)$ 可积，$y \mapsto f(x, y)$ 可导.

若 $\exists F \in L(E)$，s.t. $\Big\vert \dfrac{d}{dy} f(x, y)\Big\vert < F(x)$，则

\[
\dfrac{d}{dy} \int_E f(x, y) \, dx = \int_E \dfrac{d}{dy} f(x, y) \, dx
\]

pf. $\forall y \in (a, b)$，$h_k > 0$，$\displaystyle\lim_{k \to \infty} \frac{f(x, y + h_k) - f(x, y)}{h_k} = \dfrac{d}{dy} f(x, y)$.

注意 $\Big\vert \dfrac{f(x, y + h_k) - f(x, y)}{h_k}\Big\vert = \Big\vert \dfrac{d}{dy} f(x, y + \theta_k h_k)\Big\vert \leq F(x)$.

由 DCT 即有 $\displaystyle\int_E \lim_{k \to \infty} \frac{f(x, y + h_k) - f(x, y)}{h_k} = \lim_{k \to \infty} \int_E \frac{f(x, y + h_k) - f(x, y)}{h_k}$.

即 $\displaystyle\int_E \dfrac{d}{dy} f(x, y) \, dx = \dfrac{d}{dy} \int_E f(x, y) \, dx$（RHS by Heine）. $\square$

\secn{4. L¹ 空间的若干讨论}

\textbf{Prop.}（L¹ 中的稠密性）

设 $f \in L(E)$，则 $\forall \varepsilon > 0$，存在简单函数 $\varphi \in L(E)$，阶梯函数 $\psi \in L(E)$，紧支连续函数 $g \in L(E)$，s.t.

\[
\Vert f - \varphi\Vert < \varepsilon \ ,\ \Vert f - \psi\Vert < \varepsilon \ ,\ \Vert f - g\Vert < \varepsilon.
\]

pf.（1）$\exists$ 简单函数列 $\varphi_k \to f$ a.e. 且 $\vert \varphi_k\vert \leq \vert f\vert$.

从而 $\vert \varphi_k - f\vert \leq 2\vert f\vert \in L(E)$，由 DCT 即有 $\Vert \varphi_k - f\Vert \to 0$.

（2）不妨设 $\Vert \varphi_k - f\Vert < \dfrac{\varepsilon}{2}$，$\varphi_k = \sum\limits_{j=1}^{N} a_j 1_{A_j}$（$a_j \neq 0$）.

其中 $mA_j \leq \dfrac{1}{\vert a_j\vert} \int_E \vert \varphi_k\vert \, dm < \infty$，由 Littlewood's Prin.，

$\exists$ 有限个方体的并 $B_j$，s.t. $m(A_j \triangle B_j) < \varepsilon'$，$\forall j$.

令 $\psi_k = \sum\limits_{j=1}^{N} a_j 1_{B_j}$，则 $\psi_k$ 是阶梯函数.

\[
\Vert \psi_k - \varphi_k\Vert \leq \sum_{j=1}^{N} \vert a_j\vert \cdot \Vert 1_{A_j} - 1_{B_j}\Vert = \sum_{j=1}^{N} \vert a_j\vert \cdot m(A_j \triangle B_j) < \varepsilon' \cdot \sum_{j=1}^{N} \vert a_j\vert
\]

取适当 $\varepsilon'$ 即有 $\Vert \psi_k - f\Vert \leq \Vert \psi_k - \varphi_k\Vert + \Vert \varphi_k - f\Vert < \varepsilon$.

（3）由 Lusin Thm，$\forall \varepsilon'' > 0$，$\exists$ 连续 $g$，s.t.

\[
m\{g \neq \psi\} < \varepsilon'' \quad \& \quad \vert g\vert \leq \vert \psi\vert \leq M
\]

则 $\displaystyle\int_E \vert g - \psi\vert \, dm = \int_{\{g \neq \psi\}} \vert g - \psi\vert \, dm \ \leq\ 2M\varepsilon''$.

适当取 $\varepsilon''$ 即可. $\square$

\textbf{Prop}（平均连续性）设 $f \in L(\mathbb{R}^n)$，则 $\displaystyle\lim_{h \to 0} \int_{\mathbb{R}^n} \vert f(x+h) - f(x)\vert \, dx = 0$.

pf. $\forall \varepsilon > 0$，$\exists$ 紧支连续 $g$，s.t. $\Vert f - g\Vert < \varepsilon$.

$g$ 紧支连续 $\Rightarrow g$ 一致连续，则 $\exists \delta$，s.t. $\forall \vert h\vert < \delta$，

\[
\vert g(x + h) - g(x)\vert < \varepsilon'
\]

则 $\displaystyle\int_{\mathbb{R}^n} \vert g(x+h) - g(x)\vert \, dx = \int_{\mathrm{Supp}(g)\ \text{之}\ \delta\ \text{邻域}} \vert g(x+h) - g(x)\vert \, dx < \varepsilon' \cdot mS$.

注意 $\displaystyle\int_{\mathbb{R}^n} \vert f(x+h) - f(x)\vert \, dx \leq \int_{\mathbb{R}^n} \vert f(x+h) - g(x+h)\vert \, dx$

\[
+\ \int_{\mathbb{R}^n} \vert g(x+h) - g(x)\vert \, dx + \int_{\mathbb{R}^n} \vert g(x) - f(x)\vert \, dx
\]

即已证毕. $\square$

\textbf{Thm.} 设 $f, g, g_k \in L(E)$，则附加以下任一条件，有 $\displaystyle\int_E f g_k \to \int_E f g$.

（1）$f$ 有界 且 $g_k \xrightarrow{L^1} g$.

（2）$g_k \Rightarrow g$.

（3）$g$ 有界，$g_k$ 一致有界，$g_k \Rrightarrow g$.

pf.（1）设 $\vert f\vert < M$，则 $\displaystyle\int_E \vert fg - f g_k\vert \, dm < M \cdot \int_E \vert g - g_k\vert \, dm \ \to\ 0$.

（2）$\displaystyle\int_E \vert fg - f g_k\vert \, dm < \sup_{x \in E} \vert g_k(x) - g(x)\vert \cdot \int_E \vert f\vert \, dm \ \to\ 0$.

（3）$\forall \varepsilon$，取紧支连续 $f_\varepsilon$ s.t. $\Vert f - f_\varepsilon\Vert < \varepsilon$.

\[
\Vert fg - f g_k\Vert \leq \Vert fg - f_\varepsilon g\Vert + \Vert f_\varepsilon g - f_\varepsilon g_k\Vert + \Vert f_\varepsilon g_k - f g_k\Vert
\]

并且 $\exists \widehat{M}(\varepsilon)$ s.t. $\vert f_\varepsilon\vert \leq \widehat{M}(\varepsilon)$. 从而即已证毕. $\square$

\textbf{Prop} 设 $f \in L(E)$，若对 $\forall$ 紧支连续 $g$，$\displaystyle\int_E f g = 0$，则 $f = 0$，a.e.

pf. 设 $f'(x) = \begin{cases} f(x), & \text{if } \vert f(x)\vert \leq 1 \\ \mathrm{sgn} f(x), & \text{otherwise} \end{cases}$，$f' \in L(E)$.

$\exists$ 紧支连续 $g_k$，$\Vert g_k - f'\Vert \to 0$ 且 $\vert g_k\vert \leq 1$.

又 $\displaystyle\int_E f g_k = 0$，$\forall k$，故

\[
0 = \lim_{k \to \infty} \int_E f g_k = \int_E f \cdot f' = \int_{\{\vert f\vert \leq 1\}} f^2 + \int_{\{\vert f\vert > 1\}} \vert f\vert
\]

从而 $f = 0$  a.e. $x \in E$. $\square$

\textbf{Thm.}（Riemann–Lebesgue）

设 $f, g_k \in L([a,b])$，$g_k$ 一致有界. 若 $\exists M$，$\forall c \in [a,b]$，

\[
\int_{[a,c]} g_k \, dm \ \to\ M(c - a)
\]

则 $\displaystyle\int_{[a,b]} f g_k \to M \int_{[a,b]} f$.

pf. $\forall [\alpha, \beta] \subset [a,b]$，$\displaystyle\int_{[\alpha, \beta]} g_k \, dm \to M(\beta - \alpha)$.

设有阶梯函数 $\gamma = \sum c_j 1_{[\alpha_j, \beta_j]}$，则

\[
\int_{[a,b]} \gamma g_k = \sum c_j \int_{[\alpha_j, \beta_j]} g_k \, dm \ \to\ M \sum c_j (\beta_j - \alpha_j) = \int_{[a,b]} \gamma.
\]

则对于 $f \in L(E)$，$\forall \varepsilon > 0$，$\exists$ 阶梯函数 $\gamma_\varepsilon$，s.t. $\Vert \gamma_\varepsilon - f\Vert < \varepsilon$.

\[
\Big\vert \int_{[a,b]} f g_k - M \int_{[a,b]} f \Big\vert \ \leq\ \int_{[a,b]} \vert f - \gamma\vert \cdot \vert g_k\vert
\]

\[
+\ \Big\vert \int_{[a,b]} \gamma g_k - M \int_{[a,b]} \gamma\Big\vert + M \cdot \Vert f - \gamma_\varepsilon\Vert
\]

从而即已证毕. $\square$

\secn{5. Riemann 积分与 Lebesgue 积分的关系}

设 $f: [a,b] \to \mathbb{R}$ 有界，任给分割 $P$，记其直径为 $\vert P\vert$，有相应的 Darboux 和 $\overline{S}_P f$，$\underline{S}_P f$. 取一列 $P_i$，s.t. $\vert P_i\vert \to 0$，记

\[
\overline{\int}_a^b f(x)\,dx = \lim_{\vert P_i\vert \to 0} \overline{S}_{P_i} f \ ,\quad \underline{\int}_a^b f(x)\,dx = \lim_{\vert P_i\vert \to 0} \underline{S}_{P_i} f.
\]

则 $f \in R[a,b] \iff \overline{\int}_a^b f = \underline{\int}_a^b f$.

构造简单函数 $\overline{f}_P = \sum M_i \cdot 1_{[t_{i-1},\, t_i]}$，$\underline{f}_P = \sum m_i \cdot 1_{[t_{i-1},\, t_i]}$.

在加细分割列 $\{P_k\}$ 中，$\overline{f}_{P_k} \downarrow$，$\underline{f}_{P_k} \nearrow$，$\underline{f}_{P_k} \leq f \leq \overline{f}_{P_k}$.

定义 $\overline{f} = \lim\limits_{k \to \infty} \overline{f}_{P_k}$，$\underline{f} = \lim\limits_{k \to \infty} \underline{f}_{P_k}$，$\underline{f} \leq f \leq \overline{f}$.

\textbf{Th.} 设 $f: [a,b] \to \mathbb{R}$ 有界，则

\[
f \text{ 几乎处处连续} \iff f \in R[a,b].
\]

且此时 $f \in L[a,b]$，且 $\displaystyle\int_{[a,b]} f \, dm = \int_a^b f$.

pf. 令 $\overline{f} = \lim\limits_{\delta \to 0} \sup\limits_{(x-\delta, x+\delta)} f$，$\underline{f} = \lim\limits_{\delta \to 0} \inf\limits_{(x-\delta, x+\delta)} f$，则 $\underline{f} \leq f \leq \overline{f}$.

断言：$\overline{f} = \overline{f}$，$\underline{f} = \underline{f}$  a.e.

令 $A = \{P_k$ 的分割点$\}$，$mA = 0$. 下证 $\overline{f} = \overline{f}$，$\forall x \in [a,b] \setminus A$.

$\forall x$，$\forall \delta > 0$，$\exists$ 分割开区间 $(t_{i-1}^{(k)}, t_i^{(k)}) \subset (x - \delta, x + \delta)$.

则 $\overline{f}_{P_k}(x) \leq \sup\limits_{(x-\delta, x+\delta)} f$，令 $k \to \infty$ 即有 $\overline{f} \leq \overline{f}$.

另一方面，$\forall$ 分割开区间 $(t_{i-1}^{(k)}, t_i^{(k)}) \supset (x - \delta, x + \delta)$，$\exists \delta$.

则有 $\overline{f} \geq \overline{f}$. 于是 $\overline{f} = \overline{f}$，$\forall x \in [a,b] \setminus A$.

从而有 $f$ 几乎处处连续 $\iff \overline{f} = \underline{f}$ a.e. $\iff \overline{f} = f = \underline{f}$ a.e. $(\ast)$

由 $\overline{f}, \underline{f}$ 可测 $\Rightarrow f$ 可测，又 $f$ 有界，故 $f \in L[a,b]$.

注意 $\displaystyle\int_{[a,b]} \overline{f} \, dm = \lim_{k \to \infty} \int_{[a,b]} \overline{f}_{P_k} \, dm = \lim_{k \to \infty} \sum M_i \Delta t_i$

\[
= \lim_{k \to \infty} \overline{S}_{P_k} f = \overline{\int}_a^b f(x)\,dx
\]

故 $(\ast) \iff f \in R[a,b]$. $\square$

\secn{6. Fubini 定理}

\textbf{Problem.} $f: \mathbb{R}^{p+q} \to \mathbb{R}$ 可积，何时有

\[
\int_{\mathbb{R}^{p+q}} f \, dm = \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} f(x, y) \, dy \Big) dx
\]

Tonelli 条件. 设 $f$ 非负可测.

（1）$\mathbb{R}^q \ni y \mapsto f(x, y)$ 非负可测，a.e. $x \in \mathbb{R}^p$.

（2）$\mathbb{R}^p \ni x \mapsto \displaystyle\int_{\mathbb{R}^q} f(x, y) \, dy$ 非负可测.

（3）$\displaystyle\int_{\mathbb{R}^{p+q}} f(x, y) \, dm = \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} f(x, y) \, dy \Big) dx$

记满足上述 (1)(2)(3) 的 $f \in L^+(\mathbb{R}^{p+q})$ 构成集合 $\mathfrak{F}$.

易知 $\mathfrak{F} \neq \emptyset$. 取方体 $I_1 \subset \mathbb{R}^p$，$I_2 \subset \mathbb{R}^q$，

\[
f(x, y) = 1_{x \in I_1} \cdot 1_{y \in I_2}
\]

则易验证 $f \in \mathfrak{F}$.

\textbf{Thm.}（Tonelli）$\mathfrak{F} = L^+(\mathbb{R}^{p+q})$.

\textbf{Claim.}（1）$f \in \mathfrak{F}$，$c \in \mathbb{R}_+ \Rightarrow cf \in \mathfrak{F}$.

（2）$f, g \in \mathfrak{F} \Rightarrow f + g \in \mathfrak{F}$.

\textbf{Claim.} $f, g \in \mathfrak{F}$，$f - g \geq 0$，$g \in L^+(\mathbb{R}^{p+q})$，则 $f - g \in \mathfrak{F}$.

pf. 验证 (1)(2).

\[
\int_{\mathbb{R}^{p+q}} g \, dm = \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} g \, dy \Big) dx < \infty
\]

故 a.e. $x \in \mathbb{R}^p$，$\displaystyle\int_{\mathbb{R}^q} g(x, y)\, dy < \infty$，$y \mapsto g(x, y)$ 几乎处处有限.

则 $\displaystyle\int_{\mathbb{R}^{p+q}} (f - g)\, dm$ 几乎处处有定义.

对（3），用积分的线性性即得. $\square$

\textbf{Claim.} 设 $f_k \in \mathfrak{F}$，$f_k \nearrow f$，则 $f \in \mathfrak{F}$.

pf. By MCT，(1)(2) 成立：$\displaystyle\int_{\mathbb{R}^q} f(x, y)\, dy = \lim_{k \to \infty} \int_{\mathbb{R}^q} f_k(x, y)\, dy$.

对（3）. $\displaystyle\int_{\mathbb{R}^{p+q}} f(x, y)\, dm = \lim_{k \to \infty} \int_{\mathbb{R}^{p+q}} f_k(x, y)\, dm$

\[
= \lim_{k \to \infty} \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} f_k(x, y)\, dy \Big) dx = \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} \lim_{k \to \infty} f_k(x, y)\, dy \Big) dx
\]

\[
= \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} f(x, y)\, dy \Big) dx \quad \text{即证.} \qquad \square
\]

\textbf{Claim.} 设 $E \subset \mathbb{R}^{p+q}$ 可测，则 $1_E \in \mathfrak{F}$.

pf. Step 1. 设 $G$ 为开集，取不交半开闭方体 $\{I_j\}$，

s.t. $G = \bigcup\limits_{j=1}^{\infty} I_j$，则 $1_{\bigcup\limits_{j=1}^{\infty} I_j} \nearrow 1_G$.

从而可知 $1_G \in \mathfrak{F}$.

Step 2. 有界闭集 $F$，$1_F \in \mathfrak{F}$.

设 $F = G_1 \setminus G_2$，$G_i$ 是有界开集，$G_2 \subset G_1$.

则 $1_F = 1_{G_1} - 1_{G_2}$. 从而 $1_F \in \mathfrak{F}$.

Step 4. $F_\sigma$ 集 $F = \bigcup\limits_{k=1}^{\infty} F_k$，则 $1_{\bigcup\limits_{k=1}^{\infty} F_k} \nearrow 1_F$. 故 $1_F \in \mathfrak{F}$.

Step 5. 有限测度 $G_\delta$ 集.

设 $E_k \downarrow$，$mE_1 < +\infty$，$1_{E_k} \in \mathfrak{F} \Rightarrow 1_{\bigcap E_k} \in \mathfrak{F}$.

Step 6. 零测集. 设 $mE = 0$，

$\exists$ 开集 $G_k \downarrow$，s.t. $\lim mG_k = mE = 0$.

令 $H = \lim G_k$，则 $1_H \in \mathfrak{F}$，$E \subset H$.

又 $\displaystyle\int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} 1_H(x, y)\, dy \Big) dx = mH = 0$，

故 $\displaystyle\int_{\mathbb{R}^{p+q}} 1_E(x, y)\, dm = 0 = \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} f(x, y)\, dy \Big) dx$.

Step 7. $\forall$ 可测 $E$，$E = G \cup Z$，

其中 $G$ 是 $G_\delta$ 集，$Z$ 是零测集，$G \cap Z = \emptyset$. $\square$

根据上述 Claim，对 $\forall f \in L^+(\mathbb{R}^{p+q})$，取简单函数 $\varphi_k$，

$\varphi_k \in \mathfrak{F}$，$\varphi_k \nearrow f$. 即知 $f \in \mathfrak{F}$，故 Tonelli 定理得证.

\textbf{Thm.}（Fubini）设 $f \in L(\mathbb{R}^{p+q})$，则

（1）对于 a.e. $x \in \mathbb{R}^p$，$y \mapsto f(x, y)$ 可积.

（2）$\mathbb{R}^p \ni x \mapsto \displaystyle\int_{\mathbb{R}^q} f(x, y)\, dy$ 可积.

（3）$\displaystyle\int_{\mathbb{R}^{p+q}} f(x, y)\, dm = \int_{\mathbb{R}^p} \Big( \int_{\mathbb{R}^q} f(x, y)\, dy \Big) dx$.

\textbf{def.} 设 $f, g \in L(\mathbb{R}^n)$，记

\[
f * g(x) := \int_{\mathbb{R}^n} f(x-y)g(y)\, dy
\]

若 $f * g(x)$ 可积，a.e. $x \in \mathbb{R}^n$，则称为 $f$ 与 $g$ 之卷积.

\textbf{Thm.} $\displaystyle\int_{\mathbb{R}^n} \vert f * g\vert \ \leq\ \int_{\mathbb{R}^n} \vert f\vert \cdot \int_{\mathbb{R}^n} \vert g\vert$.

pf. 不妨设 $f$ 与 $g$ 可积，则 $\mathbb{R}^{2n} \ni (x, y) \mapsto f(x-y)g(y)$ 可测.

\[
\int_{\mathbb{R}^n} f * g = \int_{\mathbb{R}^n} \Big( \int_{\mathbb{R}^n} f(x-y)g(y)\, dy \Big) dx
\]

\[
= \int_{\mathbb{R}^{2n}} \big( f(x-y)\, g(y) \big)\, dx\, dy
\]

\[
= \int_{\mathbb{R}^n} \Big( \int_{\mathbb{R}^n} f(x-y)g(y)\, dx \Big) dy = \int_{\mathbb{R}^n} f \cdot \int_{\mathbb{R}^n} g.
\]

故对一般 $f, g \in L(\mathbb{R}^n)$，$\displaystyle\int_{\mathbb{R}^n} \vert f * g\vert \leq \int_{\mathbb{R}^n} \vert f\vert \cdot \int_{\mathbb{R}^n} \vert g\vert$. $\square$

\textbf{Prop.}（分布函数）设 $f: E \to \overline{\mathbb{R}}$ 可测，则

\[
\int_E \vert f\vert \, dm = \int_0^{+\infty} m\{\vert f\vert > \lambda\}\, d\lambda.
\]

pf. 令 $F(x, \lambda) = 1_{\{\vert f(x)\vert > \lambda\}}$，则

\[
\int_E \vert f\vert \, dm = \int_E \Big( \int_0^{+\infty} 1_{\{\vert f(x)\vert > \lambda\}}\, d\lambda \Big) dx = \int_E \Big( \int_0^{+\infty} F(x, \lambda)\, d\lambda \Big) dx
\]

\[
= \int_0^{+\infty} \Big( \int_E F(x, \lambda)\, dx \Big) d\lambda = \int_0^{+\infty} m\{\vert f\vert > \lambda\}\, d\lambda. \qquad \square
\]

\secn{1. Vitali 覆盖与单调函数可微性}

\textbf{def.}（Vitali 覆盖）

设 $E \subset \mathbb{R}^n$，称 $\beta$ 是 $E$ 的 Vitali 覆盖，若 $\forall x \in E$，$\forall \varepsilon > 0$，

$\exists B \in \beta$，s.t. $\mathrm{diam}\, B < \varepsilon$ 且 $x \in B$.

\textbf{Th.} 设 $m^*E < \infty$，$\beta$ 是 $E$ 的 Vitali 覆盖，由若干开、闭小球组成，则 $\forall \varepsilon > 0$，$\exists B_1, \cdots, B_k \in \beta$ 两两不交，s.t.

\[
m^*E + \varepsilon > \sum_{i=1}^{k} mB_i \ \geq\ m^*\Big(E \cap \big(\bigsqcup_{i=1}^{k} B_i\big)\Big) > m^*E - \varepsilon
\]

pf. $\exists$ 开集 $G \supset E$，s.t. $mG < m^*E + \varepsilon$.

不妨设 $\beta$ 中成员均是 $G$ 的子集，定义 $R(\beta) = \sup\limits_{B \subset \beta} r(B) < \infty$.

取 $B_1 \in \beta$，s.t. $r(B_1) > \tfrac{1}{2} R(\beta)$. 若 $B_1, B_2, \cdots, B_k$ 已取，$k \geq 1$，

令 $\beta_k := \{B \in \beta : B \cap B_i = \emptyset,\ 1 \leq i \leq k\}$.

取 $B_{k+1} \in \beta_k$，s.t. $r(B_{k+1}) > \tfrac{1}{2} R(\beta_k)$.

注意 $\sum mB_j \leq mG < \infty$，故 $mB_j \to 0$，则 $r(B_j) \to 0$.

$\forall B \in \beta_k$，$r(B) \leq R(\beta_k) < 2r(B_{k+1}) \to 0$，

$\exists j \geq k+1$，$B \cap B_j \neq \emptyset$. 设 $j$ 是第一个满足该式的参数，

s.t. $B \in \beta_{j-1}$，$r(B) < 2r(B_j)$.

取 $\widehat{B}_j$ 与 $B_j$ 同球心，$r(\widehat{B}_j) = 5r(B_j)$. 则 $E \setminus \bigsqcup\limits_{i=1}^{k} \overline{B_i} \subset \bigcup\limits_{j \geq k+1} \widehat{B}_j$.

\[
m^*\Big(E \setminus \bigsqcup_{i=1}^{k} B_i\Big) = m^*\Big(E \setminus \bigsqcup_{i=1}^{k} \overline{B_i}\Big) \leq m\Big(\bigcup_{j \geq k+1} \widehat{B}_j\Big) \to 0.
\]

从而即已证毕. $\square$

\textbf{Th.}（Lebesgue）

设 $f: [a,b] \to \mathbb{R}$ 单增，则 $f$ 几乎处处可微.

pf. 令 $D^+ f(x) = \overline{\lim}_{h \to 0+} \dfrac{f(x+h) - f(x)}{h} \qquad D_+ f(x) = \underline{\lim}_{h \to 0+} \dfrac{f(x+h) - f(x)}{h}$

$D^- f(x) = \overline{\lim}_{h \to 0-} \dfrac{f(x+h) - f(x)}{h} \qquad D_- f(x) = \underline{\lim}_{h \to 0-} \dfrac{f(x+h) - f(x)}{h}$

则 $D^+ f(x) \geq D_+ f(x)$，$D^- f(x) \geq D_- f(x)$.

$f$ 在 $x$ 处可微 $\iff D^+ f(x) = D_+ f(x) = D^- f(x) = D_- f(x) < \infty$.

下证 $D_- f(x) \geq D^+ f(x)$，a.e. $x \in [a,b]$.

为此，来证 $\{x \in [a,b] : D_- f(x) < D^+ f(x)\}$ 是零测的.

注意 $\{x \in [a,b] : D_- f(x) < D^+ f(x)\} = \bigcup\limits_{\substack{r, s \in \mathbb{Q}^+ \\ s < r}} \Big\{x \in [a,b] : D_- < s < r < D^+\Big\}$.

记 $D_{rs} = \{x \in [a,b] : D_- f(x) < s < r < D^+ f(x)\}$，来证 $m^*D_{rs} = 0$.

$D_- f(x) = \underline{\lim}_{h \to 0+} \dfrac{f(x-h) - f(x)}{-h} < s$，则 $\exists h_i \to 0^+$，s.t. $f(x) - f(x - h_i) < s h_i$.

则 $\{[x-h,\ x] : f(x) - f(x-h) < sh\}$ 构成 $E_{rs}$ 的 Vitali-Cover.

$\forall \varepsilon > 0$，$\exists [x_i - h_i, x_i]$ 两两不交，$i = 1, 2, \cdots, k$，s.t.

\[
\sum_{i=1}^{k} h_i < m^*E_{rs} + \varepsilon \ ,\quad m^*\Big(E_{rs} \cap \big(\bigsqcup_{i=1}^{k} [x_i - h_i, x_i]\big)\Big) > m^*E_{rs} - \varepsilon
\]

\[
\sum_{i=1}^{k} \big(f(x_i) - f(x_i - h_i)\big) < s \sum_{i=1}^{k} h_i < s\big(m^*E_{rs} + \varepsilon\big).
\]

记 $B = E_{rs} \cap \big(\bigsqcup\limits_{i=1}^{k} [x_i - h_i, x_i]\big)$，则 $\{[y,\ y+k] : f(y+k) - f(y) > kr\}$

构成 $B$ 的 Vitali Cover，则 $\exists [y_i, y_i + k_i]$ 互不相交，$i = 1, \cdots, q$，

s.t. $\displaystyle\sum_{i=1}^{q} k_i > m^*B - \varepsilon$，$\displaystyle\sum_{i=1}^{q} \big[f(y_i + k_i) - f(y_i)\big] > r \sum k_i > r \sum h_i$.

即有 $r\big(m^*E_{rs} - 2\varepsilon\big) < s\big(m^*E_{rs} + \varepsilon\big)$.

由 $\varepsilon$ 的任意性，$r \cdot m^*E_{rs} < s \cdot m^*E_{rs}$. 故 $m^*E_{rs} = 0$.

则 $f'$ 几乎处处有定义，$f'(x) \geq 0$，a.e. $x \in [a,b]$.

注意 $f'(x) = \lim\limits_{n \to \infty} n\big(f(x + \tfrac{1}{n}) - f(x)\big)$，$x \in [a,b]$，

由 Fatou's Lemma，

\[
\int_{[a,b]} f' \, dm \ \leq\ \underline{\lim}_{n \to \infty} \int_{[a,b]} n\big(f(x + \tfrac{1}{n}) - f(x)\big) \, dm
\]

\[
= \underline{\lim}_{n \to \infty} \Big( \int_b^{b + \frac{1}{n}} n f(x)\, dx - \int_a^{a + \frac{1}{n}} n f(x)\, dx \Big) = f(b) - \overline{\lim}_{n \to \infty} \int_a^{a + \frac{1}{n}} n f(x)\, dx
\]

\[
\leq f(b) - f(a).
\]

故 $f' \in L([a,b])$，则 $f'$ 几乎处处有限. 故 $f$ 几乎处处可微. $\square$

\textbf{Th.} 设 $\{f_n(x)\}$ 是 $[a,b]$ 上的递增函数，$\sum f_n(x)$ 在 $[a,b]$ 上收敛，则

\[
\dfrac{d}{dx}\Big( \sum_{n=1}^{\infty} f_n(x) \Big) = \sum_{n=1}^{\infty} \dfrac{d}{dx} f_n(x) \ ,\quad \text{a.e. } x \in [a,b].
\]

pf. 首先，$\sum\limits_{n=1}^{\infty} f_n(x)$ 在 $[a,b]$ 上递增，故几乎处处可微.

记 $R_N(x) = \sum\limits_{n=N+1}^{\infty} f_n(x)$，只需 $\lim\limits_{N \to \infty} R'_N(x) = 0$，a.e.

注意 $R'_N(x)$ 非负递减，故上述极限存在.

又 $\displaystyle\int_{[a,b]} \lim R'_N(x)\, dm = \lim_{N \to \infty} \int_{[a,b]} R'_N(x)\, dm$

\[
\leq \lim_{N \to \infty} \big( R_N(b) - R_N(a) \big) = 0.
\]

故 $\lim R'_N(x) = 0$，a.e. $\square$

\secn{2. 有界变差函数}

\textbf{def.} 设 $f: [a,b] \to \mathbb{R}$，$V_a^b f := \sup\limits_{\text{分割}\{a, X_i\}} \sum \vert f(x_{i+1}) - f(x_i)\vert$ 称为

$f$ 在 $[a,b]$ 上的全变差. 若 $V_a^b f < +\infty$，则称 $f$ 是有界变差函数. 记 $BV[a,b] = \{[a,b]$ 上的 BV 函数$\}$.

\textbf{Prop.}（1）$BV[a,b]$ 是实线性空间.

（2）单增函数是有界变差函数.

（3）Lipschitz 函数是有界变差函数.

\textbf{Th.}（Jordan 分解）

$\forall f \in BV[a,b]$，$\exists$ 单增函数 $g, h$，s.t. $f = g - h$.

pf. 取 $g(x) = \dfrac{V_a^x f + f(x)}{2}$，$h(x) = \dfrac{V_a^x f - f(x)}{2}$.

$\forall x < y$，$g(y) - g(x) = \dfrac{1}{2}\big( V_a^y f + f(y) - V_a^x f - f(x) \big)$

\[
= \dfrac{1}{2}\big( V_x^y f - (f(x) - f(y)) \big)
\]

\[
\geq \dfrac{1}{2}\big( V_x^y f - \vert f(y) - f(x)\vert \big) \ \geq\ 0.
\]

故 $g$ 是单增函数. 同理 $h$ 是单增函数. $\square$

\secn{3. Lebesgue 微分定理}

\textbf{Th}（Wiener Ball Covering）

设 $\beta$ 是 $\mathbb{R}^n$ 中的一簇开球，$c > 0$，s.t. $m\big(\bigcup\limits_{B \in \beta} B\big) > c$，

则存在有限个不交开球 $\{B_k\}_{k=1}^{K} \subset \beta$，s.t. $\displaystyle\sum_{k=1}^{K} mB_k > \dfrac{c}{3^n}$.

pf. 由内正则性，存在紧 $K \subset \bigcup\limits_{B \in \beta} B$，s.t. $mK > c$.

设 $B_1', B_2', \cdots, B_p' \in \beta$ 构成 $K$ 的有限覆盖.

不妨设 $mB_1' \geq mB_2' \geq \cdots \geq mB_p'$.

取 $B_1 = B_1'$，取 $B_2$ 为与 $B_1$ 不交的半径最大的 $B_k'$，

取 $B_3$ 为与 $B_1 \cup B_2$ 不交的半径最大的 $B_k'$.

依次类推直至终止，则可取出 $B_1, B_2, \cdots, B_K$ 两两不交.

$\forall B_j'$，取最大的与 $B_j'$ 相交的 $B_i$，则 $mB_j' \leq mB_i$（否则，在

第 $i$ 次选择中，$B_j' \cap \big(\bigcup\limits_{k=1}^{i-1} B_k\big) = \emptyset$，故应选出 $B_i = B_j'$）.

设 $\widehat{B}_i$ 是与 $B_i$ 同心的球，$r(\widehat{B}_i) = 3r(B_i)$，则 $B_j' \subset \widehat{B}_i$.

故 $K \subset \bigcup\limits_{j=1}^{p} B_j' \subset \bigcup\limits_{k=1}^{K} \widehat{B}_k$，于是即有 $\displaystyle\sum_{k=1}^{K} mB_k \geq \dfrac{c}{3^n}$. $\square$

\textbf{def}（中心 Hardy–Littlewood 极大函数）

\[
Hf(x) := \sup_{r > 0} \dfrac{1}{mBr} \int_{Br(x)} \vert f\vert \, dm \ ,\quad x \in \mathbb{R}^n \ ,\ f \in L(\mathbb{R}^n)
\]

\textbf{Prop.} $Hf(x)$ 可测.

pf. 下证 $\forall \alpha > 0$，$\{Hf > \alpha\}$ 是开集.

$x \in \{Hf > \alpha\} \iff \exists r > 0$，$\displaystyle\int_{Br(x)} \vert f\vert \, dm > \alpha\, mBr$

$\iff \exists r' > r$，$\displaystyle\int_{Br(x)} \vert f\vert \, dm > \alpha\, mBr'$

$\forall x \in \{Hf > \alpha\}$，$\forall y \in B_{r' - r}(x)$，$Br(x) \subset Br'(y)$，则

\[
\int_{Br'(y)} \vert f\vert \, dm \ \geq\ \int_{Br(x)} \vert f\vert \, dm > \alpha\, mBr'
\]

从而 $y \in \{Hf > \alpha\}$. 则 $\{Hf > \alpha\}$ 是开集. $\square$

\textbf{Prop}（Hardy–Littlewood）

$\exists c > 0$，s.t. $\forall \alpha > 0$，$m\{Hf > \alpha\} \leq \dfrac{c}{\alpha} \displaystyle\int_{\mathbb{R}^n} \vert f\vert \, dm$.

pf. $\forall x \in \{Hf > \alpha\}$，$\exists r_x$，s.t. $\displaystyle\int_{Br_x(x)} \vert f\vert \, dm > \alpha \cdot mBrx$.

从 $\{Br_x : x \in \{Hf > \alpha\}\}$ 中取两两不交的 $\{B_k\}_{k=1}^{K}$，s.t.

\[
\sum_{k=1}^{K} m\, Bk \ \geq\ \frac{m\{Hf > \alpha\}}{3^n} \qquad \text{（From Wiener.）}
\]

则 $\Vert f\Vert = \displaystyle\int_{\mathbb{R}^n} \vert f\vert \, dm \geq \int_{\bigcup Bk} \vert f\vert \, dm$

\[
> \alpha \cdot \Big( \sum_{k=1}^{K} mBk \Big) \ \geq\ \frac{\alpha}{3^n}\ m\{Hf > \alpha\}.
\]

从而原命题中 $c = 3^n$，即已证毕. $\square$

\textbf{Th.}（Lebesgue 微分定理 1）

设 $f \in L(\mathbb{R}^n)$，则

\[
\lim_{r \to 0+} \frac{1}{mBr} \int_{Br(x)} f \, dm = f(x) \ ,\quad \text{a.e. } x \in \mathbb{R}^n.
\]

称满足上式的 $x$ 为 $f$ 的 Lebesgue 点.

pf. 记 $E_r(f)x := \Big\vert \dfrac{1}{mBr} \int_{Br(x)} f \, dm - f(x)\Big\vert$，

$E_\alpha := \Big\{x \in \mathbb{R}^n : \overline{\lim}_{r \to 0}\ E_r(f)x > 2\alpha\Big\}$，下证 $mE_\alpha = 0$.

（1）设 $f$ 是紧支连续函数，则 $E_r(f)x \to 0$，$\forall x \in \mathbb{R}^n$.

（2）若 $f \in L(\mathbb{R}^n)$，则存在紧支连续 $g$，s.t. $\Vert f - g\Vert < \varepsilon$.

$\forall x \in E_\alpha$，

\[
2\alpha < \overline{\lim}_{r \to 0} E_r(f)x \ \leq\ \overline{\lim}_{r \to 0} \Big( \frac{1}{mBr} \Vert f - g\Vert + E_r(g)x + \vert g(x) - f(x)\vert \Big)
\]

（黄批：第一项对 $r$ 取 sup；第三项 $\downarrow 0$）

\[
\leq\ H(f - g) + \vert g(x) - f(x)\vert
\]

取 $E_\alpha^1 = \{x : H(f-g)(x) > \alpha\}$，$E_\alpha^2 = \{x : \vert g(x) - f(x)\vert > \alpha\}$.

\[
mE_\alpha^1 = m\{H(f-g) > \alpha\} \leq \frac{3^n}{\alpha} \Vert f - g\Vert < \frac{3^n}{\alpha}\, \varepsilon
\]

\[
mE_\alpha^2 = m\{\vert g - f\vert > \alpha\} \leq \frac{1}{\alpha} \Vert f - g\Vert < \frac{\varepsilon}{\alpha}
\]

$E_\alpha \subset E_\alpha^1 \cup E_\alpha^2$，于是 $mE_\alpha < \dfrac{3^n + 1}{\alpha}\, \varepsilon$. 由 $\varepsilon$ 任意性则 $mE_\alpha = 0$.

最后，注意

\[
\{x \in \mathbb{R}^n : \overline{\lim}\ E_r(f)x > 0\} \ \subset\ \bigcup_{k=1}^{\infty} E_{\frac{1}{k}}
\]

即已证毕. $\square$

\textbf{Th.}（Lebesgue 微分定理 2）

设 $f \in L(\mathbb{R}^n)$. 则

\[
\lim_{r \to 0} \frac{1}{mBr} \int_{Br(x)} \vert f(y) - f(x)\vert \, dy = 0 \ ,\quad \text{a.e. } x \in \mathbb{R}^n.
\]

pf. 定义 $\widetilde{E}_\alpha := \Big\{ x \in \mathbb{R}^n : \overline{\lim}\ \frac{1}{mBr} \int_{Br(x)} \vert f(y) - f(x)\vert \, dy > 3\alpha \Big\}$

\[
\widetilde{E}_{\alpha, g} := \Big\{ x \in \mathbb{R}^n : \vert f(x) - g\vert < \alpha \ ,\ \overline{\lim}\ \frac{1}{mBr} \int_{Br(x)} \vert f(y) - g\vert \, dy > 2\alpha \Big\}
\]

由 Lebesgue 微分定理 1，

\[
\overline{\lim}\ \frac{1}{mBr} \int_{Br(x)} \vert f(y) - g\vert \, dy = \vert f(x) - g\vert \ ,\quad \text{a.e. } x \in \mathbb{R}^n
\]

故 $m\widetilde{E}_{\alpha, g} = 0$，而 $\widetilde{E}_\alpha \subset \bigcup\limits_{g \in \mathbb{Q}} \widetilde{E}_{\alpha, g}$，故 $m\widetilde{E}_\alpha = 0$. $\square$

\textbf{Rem.}（1）"$f \in L(\mathbb{R}^n)$" 可替换为 "$f \in L_{loc}(\mathbb{R}^n)$"（i.e. 局部可积，即在任意开球上可积），取 $f \cdot 1_{B_k}$ 即可.

（2）$Br(x)$ 可替换为任意开（闭）球 $B \ni x$，即

\[
\lim_{\substack{mB \to 0 \\ x \in B}} \frac{1}{mB} \int_B f \, dm = f(x) \ ,\quad \text{a.e. } x \in \mathbb{R}^n.
\]

\textbf{e.g.}（$n = 1$）取 $B = [x,\ x+h]$，则

\[
\lim_{h \to 0} \frac{1}{h} \int_{[x, x+h]} f \, dm = f(x)
\]

$LHS = \lim\limits_{h \to 0} \dfrac{F(x + h) - F(x)}{h}$，其中 $F(x) := \displaystyle\int_a^x f(t)\, dt$.

则 Lebesgue 微分定理即为 $\dfrac{d}{dx} \displaystyle\int_a^x f(t)\, dt = f(x)$.

\textbf{Rem.} 考虑集函数 $F: \mathscr{A} \to \mathbb{R}$，$F(A) = \displaystyle\int_A f \, dm$（"不定积分"）.

称 $\mathscr{A}_x \subset \mathscr{A}$ 为 $x$ 的正则邻域系，若 $\exists c > 0$，s.t. $\forall A \in \mathscr{A}_x$，

$A \ni x$，存在球 $B \supset A$，s.t. $\dfrac{mA}{mB} > c$.

定义 $D_x F := \lim\limits_{\substack{A \text{正则收缩于} x \\ A \ni x}} \dfrac{F(A)}{mA}$，则成立 Lebesgue 微分定理：

\[
D_x F = f(x) \ ,\quad \text{a.e. } x \in \mathbb{R}^n
\]

pf. 设正则邻域 $A \ni x$，取 $Br(x) \supset A$，s.t. $\dfrac{mA}{mBr} > c > 0$，则

\[
\Big\vert \dfrac{F(A)}{mA} - f(x)\Big\vert = \dfrac{1}{mA} \Big\vert \int_A \big(f(y) - f(x)\big) dy\Big\vert
\]

\[
\leq \dfrac{1}{c\, mBr} \int_{Br} \vert f(y) - f(x)\vert \, dy
\]

\[
\to 0 \ ,\ r \to 0 \ ,\ \text{a.e. } x \in \mathbb{R}^n. \qquad \square
\]

\textbf{Rem.} 取 $f = 1_E$，$E \in \mathscr{A}$. 则

\[
\frac{m(Br(x) \cap E)}{m\, Br(x)} = \frac{1}{m\, Br(x)} \int_{Br(x)} 1_E \, dm \ \to\ 1_E(x) \ ,\ \text{a.e.}\ x \in \mathbb{R}^n
\]

若 $x \in \mathbb{R}^n$ s.t. 上式成立，则称 $x$ 为 $E$ 的 Lebesgue 点.

\textbf{Prop.} 设 $f \in L(\mathbb{R}^n)$，则 $\displaystyle\lim_{r \to 0} \int_{\mathbb{R}^n} \Big\vert \frac{1}{mBr} \int_{Br(x)} f \, dm - f(x)\Big\vert \, dx = 0$.

pf. $\displaystyle\int_{\mathbb{R}^n} \Big\vert \frac{1}{mBr} \int_{Br(x)} f \, dm - f(x)\Big\vert \, dx = \frac{1}{mBr} \int_{\mathbb{R}^n} \Big\vert \int_{Br(x)} (f(y) - f(x))\, dy\Big\vert \, dx$

\[
\leq \frac{1}{mBr} \int_{\mathbb{R}^n} \Big( \int_{Br(x)} \vert f(y) - f(x)\vert \, dy \Big) dx = \frac{1}{mBr} \int_{\mathbb{R}^n} \Big( \int_{Br(0)} \vert f(x+h) - f(x)\vert \, dh \Big) dx
\]

\[
= \frac{1}{mBr} \int_{Br(0)} \Big( \int_{\mathbb{R}^n} \vert f(x+h) - f(x)\vert \, dx \Big) \, dh \ \xrightarrow{(\ast)}\ 0.
\]

$(\ast)$ from $\displaystyle\lim_{h \to 0} \int_{\mathbb{R}^n} \vert f(x+h) - f(x)\vert \, dx = 0$. $\square$

\secn{4. 微积分基本定理}

设 $f \in L[a,b]$，$F(x) := \displaystyle\int_{[a,x]} f \, dm$.

积分绝对连续性：$\forall \varepsilon > 0$，$\exists \delta > 0$，$\forall A$，$mA < \delta$，s.t. $\displaystyle\int_A \vert f\vert \, dm < \varepsilon$.

特别地，取 $A = \bigsqcup\limits_{i=1}^{k} [x_i, y_i]$，则

\[
\varepsilon > \int_A \vert f\vert \, dm = \sum_{i=1}^{k} \int_{[x_i, y_i]} \vert f\vert \, dm \ \geq\ \sum_{i=1}^{k} \Big\vert \int_{[x_i, y_i]} f \, dm\Big\vert = \sum_{i=1}^{k} \vert F(y_i) - F(x_i)\vert.
\]

\textbf{def}（绝对连续）

称 $f: [a,b] \to \mathbb{R}$ 绝对连续，若 $\forall \varepsilon > 0$，$\exists \delta > 0$，$[x_i, y_i]$ 是有限个不交区间，$\sum\limits_{i=1}^{k} (y_i - x_i) < \delta$ 时就有 $\sum\limits_{i=1}^{k} \vert f(y_i) - f(x_i)\vert < \varepsilon$.

记 $AC[a,b] := \{[a,b]$ 上的绝对连续函数$\}$.

\textbf{Prop.} $AC[a,b]$ 是实线性空间.

\textbf{Prop.} $AC[a,b] \subset BV[a,b]$.

proof sketch. 取 $[a,b]$ $n$ 等分，s.t. $\dfrac{b - a}{n} < \delta$. 在每个 $[x_{i-1}, x_i]$

上因 $f \in AC[a,b]$ 故 $V_{x_{i-1}}^{x_i}(f) < 1$. 则 $V_a^b(f) \leq n$. 故 $f \in BV[a,b]$.

\textbf{Prop.} 设 $f, g \in AC[a,b]$，则 $fg \in AC[a,b]$.

proof sketch. $f, g \in AC[a,b] \subset BV[a,b]$. 从而 $f, g$ 有界，用该界进行放缩.

\textbf{Lemma.} 设 $f \in AC[a,b]$，则 $f' = 0$ a.e. $\Rightarrow f \equiv \mathrm{Const}$.

pf. 反设 $\exists c$，s.t. $f(c) \neq f(a)$. 令 $E = \{x \in (a, c) : f'(x) = 0\}$.

$\forall x \in E$，$\forall \varepsilon > 0$，$\exists \delta > 0$，s.t. $\vert h\vert < \delta$ 时，$\vert f(x+h) - f(x)\vert < \varepsilon \vert h\vert$.

故 $\{[x, x+h] : x \in E \ ,\ \vert f(x+h) - f(x)\vert < \varepsilon \vert h\vert\}$ 是 $E$ 的 Vitali Cover.

存在两两不交的 $[x_i, x_i + h_i]_{i=1}^{k}$ 覆盖 $E$，即

\[
m\big([a, c] \setminus \bigsqcup [x_i, x_i + h_i]\big) < \varepsilon'
\]

因 $f \in AC$，故 $\forall \varepsilon > 0$，$\exists \varepsilon' > 0$，当不交区间 $(x_i, y_i)_{i=1}^{n}$ 长度和 $< \varepsilon'$ 时，

\[
\sum_{i=1}^{n} \vert f(x_i) - f(y_i)\vert < \varepsilon. \quad \text{（} \varepsilon' \text{ 见黄批）}
\]

则 $0 < \vert f(c) - f(a)\vert \leq \sum\limits_{i=1}^{k} \vert f(x_i + h_i) - f(x_i)\vert + \Big( \sum\limits_{i=1}^{k-1} \vert f(x_i) - f(x_i + h_i)\vert$

\[
+\ \vert f(x_1) - f(a)\vert + \vert f(c) - f(x_k + h_k)\vert \Big)
\]

\[
= \sum_{i=1}^{k} \varepsilon h_i + \varepsilon \leq (c - a + 1)\,\varepsilon \quad \text{矛盾}. \qquad \square
\]

\textbf{Th.}（微积分基本定理，$n = 1$）

以下三条论述等价：

（1）$f \in AC[a,b]$

（2）$\exists g \in L[a,b]$，s.t. $f(x) - f(a) = \displaystyle\int_{[a,x]} g \, dm$

（3）$f$ 几乎处处可导，且 $f' \in L[a,b]$，且

\[
f(x) - f(a) = \int_a^x f'(t)\, dt.
\]

pf.（3）$\Rightarrow$（2）取 $g = f'$ 即得.

（2）$\Rightarrow$（1）由积分绝对连续性即得.

（1）$\Rightarrow$（3）$f \in AC \subset BV$，故 $f$ 几乎处处可微.

设 $f = f_+ - f_-$，$f_+, f_- \nearrow$. 则

\[
\int_{[a,b]} \vert f'\vert \, dm \ \leq\ \int_{[a,b]} \vert f_+'\vert \, dm + \int_{[a,b]} \vert f_-'\vert \, dm < \infty
\]

故 $f' \in L[a,b]$.

令 $\widetilde{f} = f(a) + \displaystyle\int_a^x f' \, dm$，则 $\widetilde{f}(a) = f(a)$，$\widetilde{f}' = f'$，a.e.

对 $(\widetilde{f} - f)' = 0$，a.e. 使用 Lemma 即得

\[
f(x) = f(a) + \int_a^x f' \, dm. \qquad \square
\]

\textbf{Prop.}（分部积分）设 $f, g \in AC[a,b]$. 则

\[
\int_{[a,b]} f'g \, dm + \int_{[a,b]} fg' \, dm = f(b)g(b) - f(a)g(a).
\]

proof sketch. $fg \in AC[a,b]$，$(fg)' = f'g + fg'$  a.e.

\textbf{Prop.} 设 $f \in AC[a,b]$，则 $V_a^b(f) = \Vert f'\Vert = \displaystyle\int_{[a,b]} \vert f'\vert \, dm$.

pf. $\forall$ 分割，$\displaystyle\sum_{i=1}^{n} \vert f(x_i) - f(x_{i-1})\vert = \sum_{i=1}^{n} \Big\vert \int_{[x_{i-1}, x_i]} f' \, dm\Big\vert \ \leq\ \int_{[a,b]} \vert f'\vert \, dm$.

另一方面，$\forall \varepsilon > 0$，$\exists$ 阶梯函数 $g = \sum\limits_{i=1}^{k} c_i 1_{[x_{i-1}, x_i]}$，

s.t. $\Vert f' - g\Vert < \varepsilon$. 再取 $G(x) = g(a) + \displaystyle\int_a^x g(t)\, dt$，

由 $V_a^b G \leq V_a^b f + V_a^b(G - f)$. 有

\[
V_a^b f \ \geq\ V_a^b G - V_a^b(G - f) = \sum_{i=1}^{k} c(x_i - x_{i-1}) - V_a^b(G - f)
\]

\[
\geq \Vert g\Vert - \Vert (G - f)'\Vert \ \geq\ \Vert g\Vert - \varepsilon \ \geq\ \Vert f'\Vert - 2\varepsilon. \qquad \square
\]

\textbf{Th.}（积分换元）

设 $f \in L(\mathbb{R}^n)$ 或非负可测，$\Omega \subset \mathbb{R}^n$ 可测，$\varphi: \Omega \to \mathbb{R}^n$ 是 $C^1$ 到像微分同胚. 则

\[
\int_{\varphi(\Omega)} f \, dm = \int_{\Omega} f(\varphi(x))\, \vert \mathrm{det}\, D_x \varphi\vert \, dx
\]

\textbf{Prop.} 设 $f \in L(\mathbb{R}^n)$，$g \in C_c^k(\mathbb{R}^n)$（i.e. 紧支连续）. 则 $f * g \in C^k(\mathbb{R}^n)$.

pf. $K = 0$ 时，$\vert f * g(x+h) - f * g(x)\vert = \Big\vert \int_{\mathbb{R}^n} \big(f(x+h-t) - f(x-t)\, g(t)\big) dt\Big\vert$

\[
\leq \int_{\mathbb{R}^n} \vert g(t)\vert \cdot \vert f(x+h-t) - f(x-t)\vert \, dt
\]

\[
\xrightarrow{g \text{紧支}}\ M \int_{\mathbb{R}^n} \vert f(x+h-t) - f(x-t)\vert \, dt \ \to\ 0 \quad (h \to 0).
\]

即 $f * g(x+h)$ 一致收敛到 $f * g$. 故 $f * g \in C^0(\mathbb{R}^n)$.

$K = 1$ 时，$g$ 紧支且 $\dfrac{\partial}{\partial x_i} g$ 紧支，$\qquad (\ast)$ from DCT

\[
\dfrac{\partial}{\partial x_i} f * g(x) = \dfrac{\partial}{\partial x_i} \int_{\mathbb{R}^n} f(y)\, g(x-y)\, dy \overset{(\ast)}{=} \int_{\mathbb{R}^n} \dfrac{\partial}{\partial x_i}\big( f(y)\, g(x-y) \big) dy
\]

\[
= \int_{\mathbb{R}^n} f(y)\, \dfrac{\partial}{\partial x_i} g(x-y)\, dy = f * \dfrac{\partial}{\partial x_i} g \quad \in C^0.
\]

By induction. $\square$

\secn{1. Lp 空间定义}

设 $0 < p < \infty$，$E \subset \mathbb{R}^n$ 是正测集，$f: E \to \overline{\mathbb{R}}$ 可测.

\[
\Vert f\Vert_p := \Big( \int_E \vert f\vert^p \, dm \Big)^{\frac{1}{p}}
\]

\[
\Vert f\Vert_\infty := \inf \{ M > 0 : \vert f\vert \leq M \ ,\ \text{a.e.} \}
\]

\[
= \inf_{mZ = 0} \sup_{x \in E \setminus Z} \vert f(x)\vert =: \operatorname*{ess\ sup}_{x \in E} \vert f\vert
\]

\[
L^p(E) := \{ f: E \to \overline{\mathbb{R}}\ \text{可测} : \Vert f\Vert_p < \infty \} \quad (0 < p \leq \infty)
\]

\textbf{Prop.} $L^p(E)$ 是线性空间.

pf.  \textcircled{1}  $\Vert cf\Vert_p = \vert c\vert \cdot \Vert f\Vert_p$，$c \in \mathbb{R}$.

故 $f \in L^p \Rightarrow cf \in L^p$.

 \textcircled{2}  $0 < p < \infty$ 时，$\forall f, g \in L^p$，

\[
\Vert f + g\Vert_p = \Big( \int_E \vert f + g\vert^p \, dm \Big)^{\frac{1}{p}} \leq \Big( \int_E \big(2 \max\{\vert f\vert, \vert g\vert\}\big)^p dm \Big)^{\frac{1}{p}}
\]

\[
= 2 \Big( \int_E \max\{\vert f\vert^p, \vert g\vert^p\} \, dm \Big)^{\frac{1}{p}}
\]

\[
\leq 2 \Big( \int_E (\vert f\vert^p + \vert g\vert^p) \, dm \Big)^{\frac{1}{p}} = 2 \big( \Vert f\Vert_p^p + \Vert g\Vert_p^p \big)^{\frac{1}{p}} < \infty
\]

若 $p = \infty$，则 $\forall f, g \in L^\infty$.

\[
\Vert f + g\Vert_\infty \ \leq\ \Vert f\Vert_\infty + \Vert g\Vert_\infty. \qquad \square
\]

\textbf{Prop.} $\forall f, g \in L^p(E)$，$\Vert f + g\Vert_p^p \leq 2^p \big( \Vert f\Vert_p^p + \Vert g\Vert_p^p \big)$.

\textbf{Prop.}（Hölder）

$1 < p < \infty$ 时，称 $\dfrac{1}{p} + \dfrac{1}{q} = 1$ 的 $q$ 为 $p$ 的 Hölder 共轭. 定义 1

与 $\infty$ 互为 Hölder 共轭.

设 $p, q$ Hölder 共轭，$f \in L^p(E)$，$g \in L^q(E)$，则 $fg \in L^1(E)$，

\[
\Vert fg\Vert_1 \ \leq\ \Vert f\Vert_p \cdot \Vert g\Vert_q.
\]

pf. 若 $p = 1$，$g = \infty$，即为

\[
\Vert f\Vert_1 \cdot \Vert g\Vert_\infty \ \geq\ M \cdot \Vert f\Vert_1 \ \geq\ \Vert fg\Vert_1.
\]

若 $\Vert f\Vert_p = 0$，则 $f = 0$  a.e.，故 $\Vert fg\Vert_1 = 0$.

设 $1 < p, q < \infty$，$A := \Vert f\Vert_p > 0$，$B = \Vert g\Vert_q > 0$.

\[
\frac{\Vert f \cdot g\Vert}{\Vert f\Vert_p \cdot \Vert g\Vert_q} = \int_E \frac{\vert f\vert}{A} \cdot \frac{\vert g\vert}{B} \, dm
\]

\[
\leq \int_E \frac{1}{p} \Big( \frac{\vert f\vert}{A} \Big)^p + \frac{1}{q} \Big( \frac{\vert g\vert}{B} \Big)^q \, dm = \frac{1}{p} + \frac{1}{q} = 1. \qquad \square
\]

\textbf{Prop.} $r < p < s$. $L^r \cap L^s \subseteq L^p \subseteq L^r + L^s$.

pf. 一方面，$\forall f \in L^p$，$\exists f_1 \in L^r$，$f_2 \in L^s$，s.t. $f = f_1 + f_2$.

即取 $f_1 = f \cdot 1_{\{\vert f\vert \geq 1\}}$，$f_2 = f \cdot 1_{\{\vert f\vert \leq 1\}}$.

另一方面，$\forall f \in L^r \cap L^s$，$\displaystyle\int_E \vert f\vert^r \, dm$，$\displaystyle\int_E \vert f\vert^s \, dm < \infty$.

则 $\vert f\vert^p \leq \vert f\vert^r + \vert f\vert^s \quad \Rightarrow \quad \int_E \vert f\vert^p \, dm < \infty$.

故 $f \in L^p$. 综上即已证毕. $\square$

\textbf{Prop.}（幂平均）

设 $0 < mE < \infty$，$0 < r < s$，$f \in L^r(E) \cap L^s(E)$，则

\[
p \mapsto \Big( \frac{\int_E \vert f\vert^p \, dm}{mE} \Big)^{\frac{1}{p}} \quad \text{单调不减}.
\]

pf. 设 $p < s$. $\Big( \dfrac{\int_E \vert f\vert^p \, dm}{mE} \Big)^{\frac{1}{p}} \leq \Big( \dfrac{\int_E \vert f\vert^s \, dm}{mE} \Big)^{\frac{1}{s}}$.

\[
\int_E \vert f\vert^p \, dm = \int_E \vert f\vert^p \cdot 1 \, dm \ \leq \Big( \int_E (\vert f\vert^p)^{\frac{s}{p}} \, dm \Big)^{\frac{p}{s}} \cdot \Big( \int_E 1 \, dm \Big)^{1 - \frac{p}{s}}
\]

\[
\Rightarrow \quad \Vert f\Vert_p \ \leq\ \Vert f\Vert_s \cdot (mE)^{\frac{1}{p} - \frac{1}{s}}. \qquad \square
\]

\textbf{Thm.} $p \geq 1$ 时，$\Vert \cdot \Vert_p$ 是一个范数，即

（1）$\Vert f\Vert_p = 0 \iff f = 0$，a.e.

（2）$\Vert cf\Vert_p = \vert c\vert \cdot \Vert f\Vert_p$

（3）$($Minkowski$)$ $\Vert f + g\Vert_p \leq \Vert f\Vert_p + \Vert g\Vert_p$.

pf.（3）$\displaystyle\int_E \vert f + g\vert^p \, dm \leq \int_E \vert f\vert \cdot \vert f + g\vert^{p-1} \, dm + \int_E \vert g\vert \cdot \vert f + g\vert^{p-1} \, dm$

\[
\leq \Vert f\Vert_p \cdot \Big( \int_E \vert f + g\vert^p \Big)^{\frac{1}{q}} + \Vert g\Vert_p \cdot \Big( \int_E \vert f + g\vert^p \Big)^{\frac{1}{q}}
\]

\[
= \big( \Vert f\Vert_p + \Vert g\Vert_p \big) \cdot \Vert f + g\Vert_p^{\frac{p}{q}}
\]

故 $\Vert f + g\Vert_p \leq \Vert f\Vert_p + \Vert g\Vert_p$. $\square$

\textbf{Prop.} 设 $0 < r < s$，若 $f \in L^r(E) \cap L^s(E)$，则

\[
p \mapsto \Vert f\Vert_p \quad \text{在}\ [r, s]\ \text{上连续}.
\]

pf. $r \leq p \leq s$，则 $\vert f\vert^P \leq \vert f\vert^r + \vert f\vert^s$，pointwise.

$\forall \{p_n\} \subset [r, s]$，$p_n \to p$. 有 $\vert f\vert^{p_n} \to \vert f\vert^P$，pointwise.

由控制收敛定理，

\[
\lim_{n \to \infty} \int_E \vert f\vert^{p_n} \, dm = \int_E \lim_{n \to \infty} \vert f\vert^{p_n} \, dm = \int_E \vert f\vert^P \, dm. \qquad \square
\]

\textbf{Prop.} 若 $\dfrac{1}{p} = \dfrac{\lambda}{r} + \dfrac{1-\lambda}{s}$，则 $\Vert f\Vert_p \leq \Vert f\Vert_r^{\lambda} \cdot \Vert f\Vert_s^{1-\lambda}$.

pf. $\displaystyle\int_E \vert f\vert^p \, dm = \int_E \vert f\vert^{\lambda p} \cdot \vert f\vert^{(1-\lambda)p} \, dm$

\[
\leq \Big( \int_E (\vert f\vert^{\lambda p})^{\frac{r}{\lambda p}} \, dm \Big)^{\frac{\lambda p}{r}} \Big( \int_E (\vert f\vert^{(1-\lambda)p})^{\frac{s}{(1-\lambda)p}} \, dm \Big)^{\frac{(1-\lambda)p}{s}}
\]

即 $\Vert f\Vert_p \leq \Vert f\Vert_r^{\lambda} \cdot \Vert f\Vert_s^{1-\lambda}$. $\square$

\textbf{Rem.} 此即 $(0, \infty) \ni \sigma \mapsto \Vert f\Vert_{\frac{1}{\sigma}}$ 是下凸函数.

故 $p \mapsto \Vert f\Vert_p$ 是内闭连续的.

\secn{2. Lp 空间的度量结构}

记 $f \sim g$，若 $f = g$ a.e.，给 $L^P$ 商掉 $\sim$，仍记为 $L^P$.

对 $f, g \in L^P(E)$，$p \geq 1$，定义度量

\[
d_p(f, g) = \Vert f - g\Vert_p
\]

s.t.（1）$d_p(f, g) = 0 \iff f = g$（in $\widetilde{L}^P$）.

（2）$d_p(f, g) = d_p(g, f)$.

（3）$d_p(f, h) \leq d_p(f, g) + d_p(g, h)$.

\textbf{Thm.} $L^P$ 空间是完备的，i.e.

只要 $\{f_n\} \subset L^P(E)$，s.t. $d_p(f_i, f_j) \to 0$，$i, j \to \infty$

就 $\exists f \in L^P(E)$，s.t. $d_p(f_n, f) \to 0$，$n \to \infty$.

pf. Lemma（Chebyshev）. $\Vert f\Vert_p \geq \varepsilon \cdot \big( m\{\vert f\vert > \varepsilon\} \big)^{\frac{1}{p}}$.

由 $\displaystyle\int_E \vert f\vert^p \, dm \geq \varepsilon^P \cdot m\{\vert f\vert > \varepsilon\}$ 即得.

于是由题设知 $f_n$ 依测度基本列 $\Rightarrow f_n$ 依测度收敛于 $f$.

由 Riesz Thm，$\exists f_{n_i} \to f$，a.e. 再由 Fatou's Lemma，

\[
\int_E \vert f\vert^P \, dm \ \leq\ \underline{\lim} \int \vert f_{n_i}\vert^P \, dm \ <\ \infty \quad \text{（有界性）}
\]

\textbf{Prop.} $0 < p < \infty$ 时，$\{$简单函数$\}$，$\{$阶梯函数$\}$，$\{C_c$ 函数$\}$

在 $L^P(E)$ 中稠密.（$\forall f \in L^P$，$\exists\ d_p$ 下的逼近$)$

\textbf{Rem.} $L^\infty$ 中，$\{C_c$ 函数$\}$ 不稠密（e.g. $f = 1$）.

\textbf{Prop.} 设 $0 < p < \infty$. $f, g \in L^P(\mathbb{R}^n)$. 则

\[
\lim_{\vert t\vert \to +\infty} \int_{\mathbb{R}^n} \vert f(x) + g(x+t)\vert^P \, dx = \int_{\mathbb{R}^n} \vert f\vert^P \, dm + \int_{\mathbb{R}^n} \vert g\vert^P \, dm
\]

pf.（1）若 $f, g \in C_c$，则 $\exists M > 0$，s.t. $\mathrm{supp}(f) \cup \mathrm{supp}(g) \subset B_M(0)$.

故 $\vert t\vert > 2M$ 时，$\vert f(x) + g(x+t)\vert^P = \vert f(x)\vert^P + \vert g(x+t)\vert^P$.

（2）对 $f, g \in L^P$，$\exists \widetilde{f}, \widetilde{g} \in C_c$，s.t.

\[
\Vert f - \widetilde{f}\Vert_p < \varepsilon \ ,\quad \Vert g - \widetilde{g}\Vert_p < \varepsilon.
\]

$p \geq 1$ 时，Mink. 成立，

\[
\Big\vert \Vert f + g_t\Vert_p - \big(\Vert f\Vert_p^P + \Vert g\Vert_p^P\big)^{\frac{1}{p}} \Big\vert
\]

\[
\leq \Big\vert \Vert f + g_t\Vert_p - \Vert \widetilde{f} + \widetilde{g}_t\Vert_p \Big\vert + \Big\vert \Vert \widetilde{f} + \widetilde{g}_t\Vert_p - \big(\Vert \widetilde{f}\Vert_p^P + \Vert \widetilde{g}\Vert_p^P\big)^{\frac{1}{p}} \Big\vert
\]

\[
+\ \Big\vert \big(\Vert \widetilde{f}\Vert_p^P + \Vert \widetilde{g}\Vert_p^P\big)^{\frac{1}{p}} - \big(\Vert f\Vert_p^P + \Vert g\Vert_p^P\big)^{\frac{1}{p}} \Big\vert
\]

\[
\overset{\text{Mink.}}{\leq} \Vert f - \widetilde{f}\Vert_p + \Vert g_t - \widetilde{g}_t\Vert_p + \Big\vert \Vert \widetilde{f} + \widetilde{g}_t\Vert_p - \big(\Vert \widetilde{f}\Vert_p^P + \Vert \widetilde{g}\Vert_p^P\big)^{\frac{1}{p}} \Big\vert
\]

\[
+\ \big( \Vert f - \widetilde{f}\Vert_p^P + \Vert g - \widetilde{g}\Vert_p^P \big)^{\frac{1}{p}}
\]

\[
< \varepsilon + \varepsilon + \varepsilon' + (\varepsilon^P + \varepsilon^P)^{\frac{1}{p}} \ \to\ 0.
\]

$0 < p < 1$ 时，$a^P + b^P > (a + b)^P$，$a, b > 0$.

\[
\Big\vert \int \vert f + g_t\vert^P - \int \vert f\vert^P - \int \vert g\vert^P \Big\vert
\]

\[
\leq \Big\vert \int \vert f + g_t\vert^P - \int \vert \widetilde{f} + \widetilde{g}_t\vert^P \Big\vert + \Big\vert \int \vert \widetilde{f} + \widetilde{g}_t\vert^P - \int \vert \widetilde{f}\vert^P - \int \vert \widetilde{g}\vert^P \Big\vert
\]

\[
+\ \Big\vert \int \vert \widetilde{f}\vert^P + \int \vert \widetilde{g}\vert^P - \int \vert f\vert^P - \int \vert g\vert^P \Big\vert
\]

\[
\leq \int \vert f - \widetilde{f}\vert^P + \int \vert g - \widetilde{g}\vert^P + \Big\vert \int \vert \widetilde{f} + \widetilde{g}_t\vert^P - \int \vert \widetilde{f}\vert^P - \int \vert \widetilde{g}\vert^P \Big\vert
\]

\[
+\ \int \vert \widetilde{f} - f\vert^P + \int \vert \widetilde{g} - g\vert^P
\]

\[
< 4\varepsilon + \varepsilon' \ \to\ 0. \qquad \square
\]

\textbf{def.} 称 $\Gamma \subset L^P(E)$ 稠密，若 $\forall \varepsilon > 0$，$\forall f \in L^P(E)$，$\exists g \in \Gamma$，

s.t. $\Vert f - g\Vert_p < \varepsilon$.

称 $L^P(E)$ 可分，若 $L^P(E)$ 存在一个可数稠密子集.

\textbf{Prop.} 设 $E \in \mathscr{A}$，$mE > 0$，则

$0 < p < \infty$ 时，$L^P(E)$ 可分；$p = \infty$ 时，$L^P(E)$ 不可分.

pf. $0 < p < \infty$ 时，构造

\[
\Gamma = \Big\{ \sum_{i=1}^{k} c_i \cdot 1_{B_i}\ \Big|\ c_i \in \mathbb{Q} \ ,\ B_i \text{ 是有理端点方体} \Big\}.
\]

则 $\Gamma \subset L^\infty(E) \cap L^P(E)$，$\forall p > 0$.

易证 $\Gamma$ 在 $L^P(E)$ 中可数稠密.

\textbf{Lemma.} 设 $f \in L^P(E)$，$1 \leq p \leq \infty$，则

\[
\Vert f\Vert_p = \sup_{0 \neq g \in L^q} \frac{\vert \int_E f g \, dm\vert}{\Vert g\Vert_q} \quad \text{（} p, q\ \text{Hölder 共轭} \text{）}
\]

pf. 由 Hölder，$\Vert f\Vert_p \geq \dfrac{\vert \int_E f g \, dm\vert}{\Vert g\Vert_q}$. 只需验证取等.

$1 < p < \infty$ 时，取 $g = \vert f\vert^{p-1}\, \mathrm{sgn}(f)$.

$p = 1$ 时，取 $g = \mathrm{sgn}(f)$.

$p = \infty$ 时，$\forall \varepsilon > 0$，记 $A = \{\vert f\vert > \Vert f\Vert_\infty - \varepsilon\}$，$mA > 0$.

取 $g = 1_A\, \mathrm{sgn}(f)$，则

\[
\frac{\int_E f g \, dm}{\Vert g\Vert_1} = \frac{\int_A \vert f\vert \, dm}{mA} > \Vert f\Vert_\infty - \varepsilon. \qquad \square
\]

\textbf{Prop.} 设 $f: E \to \overline{\mathbb{R}}$ 可测（几乎处处有限），若 $\exists M > 0$，s.t.

$\forall g \in L^q(E)$，都有 $\displaystyle\int_E \vert f g\vert \, dm \leq M \Vert g\Vert_q$，则

\[
f \in L^P(E) \quad \text{且} \quad \Vert f\Vert_p \leq M.
\]

pf. 取非负简单紧支 $g_k \nearrow \vert f\vert^{p-1}$  a.e.，则

\[
M \cdot \Vert g_k\Vert_q \ \geq\ \int_E \vert f g_k\vert \, dm \ \geq\ \int_E g_k^{\frac{p}{p-1}} \, dm = \int_E g_k^q \, dm
\]

故 $\displaystyle\int_E \vert f\vert^p \, dm = \lim_{k \to \infty} \int_E \vert g_k\vert^q \, dm \leq M^P$. $\square$

\textbf{Prop.}（广义 Minkowski）

设 $f: \mathbb{R}^n \times \mathbb{R}^n \to \mathbb{R}$，若 $f$ 非负可测 或 可积，则

\[
\Big( \int_{\mathbb{R}^n} \Big( \int_{\mathbb{R}^n} \vert f(x, y)\vert \, dy \Big)^P dx \Big)^{\frac{1}{p}} \ \leq\ \int_{\mathbb{R}^n} \Big( \int_{\mathbb{R}^n} \vert f(x, y)\vert^P \, dx \Big)^{\frac{1}{p}} \, dy
\]

\textbf{Th.} $0 < p < \infty$. 设 $f_j \to f$ a.e. 且 $\displaystyle\int_E \vert f_j\vert^p \, dm \leq M$，则

\[
\lim_{j \to \infty} \int_E \big( \vert f_j\vert^p - \vert f\vert^p - \vert f_j - f\vert^p \big) \, dm = 0.
\]

\textbf{Th.} 设 $f \in L^P(E)$，$f_j \to f$，则

\[
\lim_{j \to \infty} \int_E \vert f_j\vert^p \, dm = \int_E \vert f\vert^p \, dm \iff \int_E \vert f - f_j\vert^p \, dm \to 0
\]

\secn{3. 恒同逼近（AI）}

$K_\varepsilon \in L(\mathbb{R}^n)$，$\int K_\varepsilon = 1$，$\forall \varepsilon > 0$.

\[
\int_{\mathbb{R}^n} \vert K_\varepsilon\vert \, dm \leq C \ ,\quad \forall \delta > 0 \ ,\ \lim_{\varepsilon \to 0+} \int_{\vert x\vert > \delta} \vert K_\varepsilon\vert \, dm = 0
\]

称满足上述条件的 $\{K_\varepsilon\}$ 为 AI.

\textbf{Prop.} 设 $f$ 有界可测. $x$ 是 $f$ 的连续点，则 $K_\varepsilon \ast f(x) \xrightarrow{\varepsilon \to 0+} f(x)$.

pf. $\vert K_\varepsilon \ast f(x) - f(x)\vert = \Big\vert \int_{\mathbb{R}^n} f(x-y)K_\varepsilon(y)\, dy - \int_{\mathbb{R}^n} f(x)K_\varepsilon(y)\, dy\Big\vert$

\[
\leq \int_{\mathbb{R}^n} \vert f(x-y) - f(x)\vert \cdot \vert K_\varepsilon(y)\vert \, dy = \int_{\vert y\vert > \delta} + \int_{\vert y\vert \leq \delta}
\]

（黄批：$x$ 是连续点，用于 $\int_{\vert y\vert \leq \delta}$）

\[
\leq 2M \int_{\vert y\vert > \delta} \vert K_\varepsilon(y)\vert \, dy + \varepsilon' \int_{\vert y\vert \leq \delta} \vert K_\varepsilon(y)\vert \, dy \ \to\ 0. \qquad \square
\]

\textbf{Prop.} 设 $f \in L^P(\mathbb{R}^n)$，$1 \leq p < \infty$，则 $\Vert K_\varepsilon \ast f - f\Vert_p \to 0$，$\varepsilon \to 0+$.

pf. $\Vert K_\varepsilon \ast f - f\Vert_p = \Big( \int_{\mathbb{R}^n} \vert K_\varepsilon \ast f - f\vert^P \, dm \Big)^{\frac{1}{p}}$

\[
\leq \Big( \int_{\mathbb{R}^n} \Big( \int_{\mathbb{R}^n} \vert K_\varepsilon(y)\vert \cdot \vert f(x-y) - f(x)\vert \, dy \Big)^P dx \Big)^{\frac{1}{p}}
\]

\[
\overset{\text{Mink.}}{\leq} \int_{\mathbb{R}^n} \Big( \int_{\mathbb{R}^n} \vert K_\varepsilon(y)\vert^P \cdot \vert f(x-y) - f(x)\vert^P \, dx \Big)^{\frac{1}{p}} \, dy
\]

\[
= \int_{\mathbb{R}^n} \vert K_\varepsilon(y)\vert \Big( \int_{\mathbb{R}^n} \vert f(x-y) - f(x)\vert^P \, dx \Big)^{\frac{1}{p}} \, dy
\]

（黄批：积分平均连续）

\[
\leq \int_{\vert y\vert > \delta} \vert K_\varepsilon(y)\vert \cdot 2\Vert f\Vert_p \, dy + \int_{\vert y\vert \leq \delta} \vert K_\varepsilon(y)\vert \cdot \varepsilon' \, dy
\]

\[
\to\ 0. \qquad \square
\]

称 $\{K_\varepsilon\}$ 是加强 AI，若 $\displaystyle\int K_\varepsilon = 1$，$\vert K_\varepsilon(x)\vert \leq \min\Big\{ \dfrac{C}{\varepsilon^n} \ ,\ \dfrac{C\varepsilon^a}{\vert x\vert^{n+a}} \Big\}$.

\textbf{Th.} 设 $f \in L(\mathbb{R}^n)$，$x$ 是 $f$ 的 Lebesgue 点，则 $K_\varepsilon \ast f(x) \to f(x)$，

$\varepsilon \to 0+$，故 $K_\varepsilon \ast f \to f$  a.e.

pf. 对 Lebesgue 点 $x$，$I(r) := \dfrac{1}{r^n} \displaystyle\int_{Br(x)} \vert f(x-y) - f(x)\vert \, dy$.

则 $I(r) \to 0$，$r \to 0+$，$r \to +\infty$，$I(r) \in C[0, +\infty)$，$I(r)$ 有界.  $(\ast)$

\[
\vert K_\varepsilon \ast f(x) - f(x)\vert \leq \int_{\mathbb{R}^n} \vert f(x-y) - f(x)\vert \cdot \vert K_\varepsilon(y)\vert \, dy
\]

\[
= \int_{\vert y\vert \leq \varepsilon} + \sum_{k=0}^{\infty} \int_{2^k \varepsilon < \vert y\vert \leq 2^{k+1} \varepsilon}
\]

\[
\leq \frac{C}{\varepsilon^n} \int_{\vert y\vert \leq \varepsilon} \vert f(x-y) - f(x)\vert \, dy + \sum_{k=0}^{\infty} \frac{C\varepsilon^a}{(2^k \varepsilon)^{n+a}} \int_{2^k \varepsilon < \vert y\vert \leq 2^{k+1} \varepsilon} \vert f(x-y) - f(x)\vert \, dy
\]

\[
\leq C\, I(\varepsilon) + \sum_{k=0}^{\infty} \frac{2^n C}{2^{ka}} \cdot \frac{1}{(2^{k+1}\varepsilon)^n} \int_{\vert y\vert \leq 2^{k+1} \varepsilon} \vert f(x-y) - f(x)\vert \, dy
\]

\[
= C\, I(\varepsilon) + \widetilde{C} \sum_{k=0}^{N} I(2^{k+1}\varepsilon) + 2^n C M \sum_{k > N} \frac{1}{2^{ka}} \quad (\ast)
\]

\[
\to\ 0. \qquad \square
\]

\textbf{Prop.} 设 $K \in L(\mathbb{R}^n)$，$K_\varepsilon(x) := \dfrac{1}{\varepsilon^n} K\Big( \dfrac{x}{\varepsilon} \Big)$，$\varepsilon > 0$. $\displaystyle\int K \, dm = 1$.

验证 $\displaystyle\int K_\varepsilon(x)\, dx = \int K\Big(\dfrac{x}{\varepsilon}\Big) d\Big(\dfrac{x}{\varepsilon}\Big) = 1$，$\displaystyle\int \vert K_\varepsilon(x)\vert \, dx = \int \vert K(x)\vert \, dx = C$.

$\forall \delta > 0$，$\displaystyle\int_{\vert x\vert \geq \delta} K_\varepsilon(x)\, dx = \int_{\vert y\vert \geq \frac{\delta}{\varepsilon}} K(y)\, dy \ \to\ 0$，$\varepsilon \to 0+$.

故 $\{K_\varepsilon\}$ 是 AI.

\textbf{Prop.} 设 $K \in L^\infty(\mathbb{R}^n)$，$\displaystyle\int K \, dm = 1$，$\vert K(x)\vert \leq \dfrac{C}{\vert x\vert^{n+a}}$.

则 $\{K_\varepsilon\}$ 是加强 AI.

\textbf{e.g.} $K(x) := \begin{cases} C \cdot e^{-\frac{1}{1 - \vert x\vert^2}}, & \vert x\vert < 1 \\ 0, & \vert x\vert \geq 1 \end{cases}$，其中 $C$ s.t. $\displaystyle\int K \, dm = 1$.

$K \in C_c^{\infty}(\mathbb{R}^n)$ 紧支光滑. 由 $K$ 生成的 $K_\varepsilon$ s.t.

\[
K_\varepsilon \ast f \to f \ ,\ \text{a.e.}\ (f \in L) \ .\quad K_\varepsilon \ast f \to f\ \text{in}\ L^P \ ,\ (f \in L^P)
\]

\textbf{Th.} $\{$紧支光滑函数$\}$ 在 $L^P$（$1 \leq p < \infty$）中稠密.

\textbf{Prop.} 设 $\Gamma \subset L^q$（$1 \leq p < \infty$）稠密，若 $\exists M > 0$，s.t. $\forall \varphi \in \Gamma$，

\[
\int \vert f \varphi\vert \, dm \leq M \cdot \Vert \varphi\Vert_q \ ,\quad \text{则}\ f \in L^P(\mathbb{R}^n) \ ,\ \Vert f\Vert_p \leq M
\]

\textbf{Rem.} 若 $\Gamma \subset L^q$ 稠密，则 $\Vert f\Vert_p = \sup\limits_{\varphi \in \Gamma} \dfrac{\int \vert f \varphi\vert \, dm}{\Vert \varphi\Vert_q}$.
